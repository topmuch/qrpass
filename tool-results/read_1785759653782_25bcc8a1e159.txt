     1→'use client';
     2→
     3→import { useState, useEffect, useRef } from 'react';
     4→import Link from 'next/link';
     5→import { QRCodeSVG } from 'qrcode.react';
     6→import {
     7→  Search,
     8→  Eye,
     9→  Download,
    10→  Share2,
    11→  Trash2,
    12→  Plane,
    13→  Luggage,
    14→  QrCode,
    15→  X,
    16→  AlertTriangle,
    17→  RefreshCw,
    18→  FileText,
    19→  Building2,
    20→  ChevronDown,
    21→  ChevronRight
    22→} from "lucide-react";
    23→
    24→interface QRSet {
    25→  id: string;
    26→  setId: string;
    27→  type: string;
    28→  agencyId: string | null;
    29→  agencyName: string | null;
    30→  createdAt: Date;
    31→  qrCount: number;
    32→  references: string[];
    33→  status: string;
    34→  travelerName: string | null;
    35→  activationStatus: 'new' | 'partial' | 'activated';
    36→  baggageIds: string[];
    37→}
    38→
    39→interface AgencyGroup {
    40→  agencyId: string | null;
    41→  agencyName: string;
    42→  sets: QRSet[];
    43→  totalQr: number;
    44→  isExpanded: boolean;
    45→}
    46→
    47→interface Stats {
    48→  totalSets: number;
    49→  totalQr: number;
    50→  hajjSets: number;
    51→}
    52→
    53→export default function EtiquettesPage() {
    54→  const qrRef = useRef<HTMLDivElement>(null);
    55→
    56→  const [sets, setSets] = useState<QRSet[]>([]);
    57→  const [agencyGroups, setAgencyGroups] = useState<AgencyGroup[]>([]);
    58→  const [stats, setStats] = useState<Stats>({
    59→    totalSets: 0,
    60→    totalQr: 0,
    61→    hajjSets: 0,
    62→
    63→  });
    64→  const [loading, setLoading] = useState(true);
    65→  const [search, setSearch] = useState('');
    66→  const [activeTab, setActiveTab] = useState<'hajj'>('hajj');
    67→
    68→  // Modals
    69→  const [showDetailModal, setShowDetailModal] = useState(false);
    70→  const [showDeleteModal, setShowDeleteModal] = useState(false);
    71→  const [selectedSet, setSelectedSet] = useState<QRSet | null>(null);
    72→  const [isDownloading, setIsDownloading] = useState(false);
    73→
    74→  useEffect(() => {
    75→    fetchSets();
    76→  }, [activeTab, search]);
    77→
    78→  const fetchSets = async () => {
    79→    setLoading(true);
    80→    try {
    81→      const params = new URLSearchParams();
    82→      params.set('type', activeTab);
    83→      if (search) params.set('search', search);
    84→
    85→      const response = await fetch(`/api/qrcodes?${params}`);
    86→      const data = await response.json();
    87→
    88→      // Calculate activation status for each set
    89→      const setsWithStatus = data.sets.map((set: QRSet) => ({
    90→        ...set,
    91→        activationStatus: getActivationStatus(set.status)
    92→      }));
    93→
    94→      // Filter to only show sets matching the active tab type
    95→      const filteredSets = setsWithStatus.filter((set: QRSet) => set.type === activeTab);
    96→      
    97→      setSets(filteredSets);
    98→      setStats(data.stats);
    99→
   100→      // Group by agency
   101→      const groupedByAgency = groupByAgency(filteredSets);
   102→      setAgencyGroups(groupedByAgency);
   103→    } catch (error) {
   104→      console.error('Error fetching QR sets:', error);
   105→    } finally {
   106→      setLoading(false);
   107→    }
   108→  };
   109→
   110→  const groupByAgency = (sets: QRSet[]): AgencyGroup[] => {
   111→    const groups = new Map<string | null, AgencyGroup>();
   112→
   113→    sets.forEach(set => {
   114→      const key = set.agencyId || 'no-agency';
   115→      const agencyName = set.agencyName || 'Sans agence';
   116→
   117→      if (!groups.has(key)) {
   118→        groups.set(key, {
   119→          agencyId: set.agencyId,
   120→          agencyName,
   121→          sets: [],
   122→          totalQr: 0,
   123→          isExpanded: true, // Expanded by default
   124→        });
   125→      }
   126→
   127→      const group = groups.get(key)!;
   128→      group.sets.push(set);
   129→      group.totalQr += set.qrCount;
   130→    });
   131→
   132→    // Sort: agencies with name first, then "Sans agence"
   133→    return Array.from(groups.values()).sort((a, b) => {
   134→      if (a.agencyId === null) return 1;
   135→      if (b.agencyId === null) return -1;
   136→      return a.agencyName.localeCompare(b.agencyName);
   137→    });
   138→  };
   139→
   140→  const getActivationStatus = (status: string): 'new' | 'partial' | 'activated' => {
   141→    if (status === 'active' || status === 'scanned') return 'activated';
   142→    if (status === 'partial') return 'partial';
   143→    return 'new';
   144→  };
   145→
   146→  const toggleAgencyGroup = (agencyId: string | null) => {
   147→    setAgencyGroups(prev => 
   148→      prev.map(group => 
   149→        group.agencyId === agencyId 
   150→          ? { ...group, isExpanded: !group.isExpanded }
   151→          : group
   152→      )
   153→    );
   154→  };
   155→
   156→  const handleDeleteSet = async () => {
   157→    if (!selectedSet) return;
   158→
   159→    try {
   160→      const params = new URLSearchParams({ setId: selectedSet.setId });
   161→      const response = await fetch(`/api/qrcodes?${params}`, { method: 'DELETE' });
   162→      const data = await response.json();
   163→
   164→      if (response.ok && data.success) {
   165→        fetchSets();
   166→        setShowDeleteModal(false);
   167→        setSelectedSet(null);
   168→      } else {
   169→        alert(`Erreur: ${data.error || 'Erreur lors de la suppression'}`);
   170→      }
   171→    } catch (error) {
   172→      console.error('Error deleting set:', error);
   173→      alert('Erreur lors de la suppression');
   174→    }
   175→  };
   176→
   177→  const handleDownloadPDF = async (set: QRSet) => {
   178→    setIsDownloading(true);
   179→    setSelectedSet(set);
   180→
   181→    try {
   182→      const canvas = document.createElement('canvas');
   183→      const ctx = canvas.getContext('2d');
   184→      if (!ctx) throw new Error('Canvas not supported');
   185→
   186→      const qrSize = 300;
   187→      const padding = 40;
   188→      const headerHeight = 100;
   189→
   190→      const cols = Math.min(set.qrCount, 3);
   191→      const rows = Math.ceil(set.qrCount / 3);
   192→
   193→      canvas.width = cols * qrSize + (cols + 1) * padding;
   194→      canvas.height = headerHeight + rows * qrSize + (rows + 1) * padding + 60;
   195→
   196→      // Background
   197→      ctx.fillStyle = '#ffffff';
   198→      ctx.fillRect(0, 0, canvas.width, canvas.height);
   199→
   200→      // Header
   201→      ctx.fillStyle = set.type === 'hajj' ? '#1D4ED8' : '#f59e0b';
   202→      ctx.fillRect(0, 0, canvas.width, headerHeight);
   203→
   204→      ctx.fillStyle = '#ffffff';
   205→      ctx.font = 'bold 24px Arial';
   206→      ctx.textAlign = 'center';
   207→      ctx.fillText('QRPass - Étiquettes', canvas.width / 2, 35);
   208→
   209→      ctx.font = '16px Arial';
   210→      ctx.fillText(`${set.setId} | Hajj 2026 | ${set.qrCount} QR`, canvas.width / 2, 65);
   211→      if (set.agencyName) {
   212→        ctx.font = '14px Arial';
   213→        ctx.fillText(`Agence: ${set.agencyName}`, canvas.width / 2, 85);
   214→      }
   215→
   216→      // Generate QR codes
   217→      for (let i = 0; i < set.references.length; i++) {
   218→        const col = i % 3;
   219→        const row = Math.floor(i / 3);
   220→        const x = padding + col * (qrSize + padding);
   221→        const y = headerHeight + padding + row * (qrSize + padding);
   222→
   223→        ctx.fillStyle = '#ffffff';
   224→        ctx.fillRect(x, y, qrSize, qrSize);
   225→
   226→        const qrUrl = `${window.location.origin}/found/${set.references[i]}`;
   227→        const svgElement = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
   228→        const qrSvg = <QRCodeSVG value={qrUrl} size={qrSize - 40} level="H" fgColor={set.type === 'hajj' ? '#1D4ED8' : '#f59e0b'} />;
   229→
   230→        // Use a promise-based approach to render SVG to canvas
   231→        const svgString = `<svg xmlns="http://www.w3.org/2000/svg" width="${qrSize - 40}" height="${qrSize - 40}">
   232→          <rect width="100%" height="100%" fill="white"/>
   233→          ${generateQRPath(qrUrl, qrSize - 40, set.type === 'hajj' ? '#1D4ED8' : '#f59e0b')}
   234→        </svg>`;
   235→
   236→        const img = new Image();
   237→        const svgBlob = new Blob([svgString], { type: 'image/svg+xml;charset=utf-8' });
   238→        const url = URL.createObjectURL(svgBlob);
   239→
   240→        await new Promise<void>((resolve) => {
   241→          img.onload = () => {
   242→            ctx.drawImage(img, x + 20, y + 20, qrSize - 40, qrSize - 40);
   243→            URL.revokeObjectURL(url);
   244→            resolve();
   245→          };
   246→          img.onerror = () => {
   247→            URL.revokeObjectURL(url);
   248→            resolve();
   249→          };
   250→          img.src = url;
   251→        });
   252→
   253→        // Reference label
   254→        ctx.fillStyle = '#1e293b';
   255→        ctx.font = 'bold 12px Arial';
   256→        ctx.textAlign = 'center';
   257→        ctx.fillText(set.references[i], x + qrSize / 2, y + qrSize - 15);
   258→      }
   259→
   260→      // Footer
   261→      ctx.fillStyle = '#64748b';
   262→      ctx.font = '12px Arial';
   263→      ctx.textAlign = 'center';
   264→      ctx.fillText(`Généré le ${new Date().toLocaleDateString('fr-FR')} | QRPass.com`, canvas.width / 2, canvas.height - 20);
   265→
   266→      // Download
   267→      const link = document.createElement('a');
   268→      link.download = `QRPass-${set.setId}.png`;
   269→      link.href = canvas.toDataURL('image/png');
   270→      link.click();
   271→
   272→      setIsDownloading(false);
   273→    } catch (error) {
   274→      console.error('Error generating PDF:', error);
   275→      setIsDownloading(false);
   276→      alert('Erreur lors de la génération');
   277→    }
   278→  };
   279→
   280→  // Simple QR path generator fallback
   281→  const generateQRPath = (url: string, size: number, color: string): string => {
   282→    return `<text x="50%" y="50%" text-anchor="middle" fill="${color}" font-size="10">${url}</text>`;
   283→  };
   284→
   285→  const handleShareSet = async (set: QRSet) => {
   286→    const shareUrl = `${window.location.origin}/found/${set.references[0]}`;
   287→    if (navigator.share) {
   288→      try {
   289→        await navigator.share({
   290→          title: `QRPass - ${set.setId}`,
   291→          text: `${set.qrCount} QR codes pour ${set.agencyName || 'agence'}`,
   292→          url: shareUrl,
   293→        });
   294→      } catch (err) {
   295→        console.log('Share cancelled');
   296→      }
   297→    } else {
   298→      navigator.clipboard.writeText(shareUrl);
   299→      alert('Lien copié !');
   300→    }
   301→  };
   302→
   303→  const formatDate = (date: Date | string) => {
   304→    return new Date(date).toLocaleDateString('fr-FR', {
   305→      day: '2-digit',
   306→      month: 'short',
   307→      year: 'numeric',
   308→    });
   309→  };
   310→
   311→  return (
   312→    <div className="max-w-7xl mx-auto">
   313→      {/* Header */}
   314→      <div className="flex flex-col md:flex-row md:items-center justify-between gap-4 mb-6">
   315→        <div>
   316→          <h1 className="text-2xl font-bold text-slate-800 dark:text-white">Étiquettes QR</h1>
   317→          <p className="text-slate-500 dark:text-slate-400 text-sm">
   318→            {stats.totalSets} sets • {stats.totalQr} QR codes
   319→          </p>
   320→        </div>
   321→
   322→        <div className="flex items-center gap-3">
   323→          <div className="relative">
   324→            <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" />
   325→            <input
   326→              type="text"
   327→              placeholder="Rechercher..."
   328→              value={search}
   329→              onChange={(e) => setSearch(e.target.value)}
   330→              className="pl-10 pr-4 py-2 w-64 bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-xl text-sm focus:outline-none focus:ring-2 focus:ring-green-500"
   331→            />
   332→          </div>
   333→
   334→          <button
   335→            onClick={() => fetchSets()}
   336→            className="p-2 bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-xl hover:bg-slate-50 dark:hover:bg-slate-700 transition-colors"
   337→          >
   338→            <RefreshCw className={`w-4 h-4 text-slate-500 ${loading ? 'animate-spin' : ''}`} />
   339→          </button>
   340→
   341→          <Link
   342→            href="/admin/generer"
   343→            className="flex items-center gap-2 px-4 py-2 bg-green-500 hover:bg-green-600 text-white rounded-xl font-medium transition-colors"
   344→          >
   345→            <QrCode className="w-4 h-4" />
   346→            Générer QR
   347→          </Link>
   348→        </div>
   349→      </div>
   350→
   351→      {/* Tabs */}
   352→      <div className="flex gap-2 mb-6">
   353→        <button
   354→          onClick={() => setActiveTab('hajj')}
   355→          className={`flex items-center gap-2 px-6 py-3 rounded-xl font-medium transition-all ${
   356→            activeTab === 'hajj'
   357→              ? 'bg-blue-700 text-white shadow-lg shadow-blue-700/30'
   358→              : 'bg-white dark:bg-slate-800 text-slate-600 dark:text-slate-300 border border-slate-200 dark:border-slate-700 hover:bg-slate-50 dark:hover:bg-slate-700'
   359→          }`}
   360→        >
   361→          <Luggage className="w-5 h-5" />
   362→          Hajj
   363→          <span className={`px-2 py-0.5 rounded-full text-xs ${
   364→            activeTab === 'hajj' 
   365→              ? 'bg-white/20 text-white' 
   366→              : 'bg-slate-100 dark:bg-slate-700 text-slate-600 dark:text-slate-400'
   367→          }`}>
   368→            {stats.hajjSets}
   369→          </span>
   370→        </button>
   371→      </div>
   372→
   373→      {/* Content */}
   374→      {loading ? (
   375→        <div className="flex items-center justify-center py-20">
   376→          <div className="animate-spin w-8 h-8 border-4 border-slate-200 border-t-green-500 rounded-full" />
   377→        </div>
   378→      ) : agencyGroups.length === 0 ? (
   379→        <div className="bg-white dark:bg-slate-800 rounded-2xl border border-slate-200 dark:border-slate-700 p-12 text-center">
   380→          <div className={`w-16 h-16 mx-auto mb-4 rounded-2xl flex items-center justify-center ${
   381→            activeTab === 'hajj' ? 'bg-emerald-100 dark:bg-blue-600/20' : 'bg-amber-100 dark:bg-blue-600/20'
   382→          }`}>
   383→            {activeTab === 'hajj' ? (
   384→              <Luggage className="w-8 h-8 text-blue-700" />
   385→            ) : (
   386→              <Plane className="w-8 h-8 text-amber-600" />
   387→            )}
   388→          </div>
   389→          <h3 className="text-lg font-semibold text-slate-800 dark:text-white mb-2">
   390→            Aucun QR code Hajj
   391→          </h3>
   392→          <p className="text-slate-500 dark:text-slate-400 mb-4">
   393→            {search ? 'Aucun résultat pour votre recherche' : 'Commencez par générer des QR codes'}
   394→          </p>
   395→          {!search && (
   396→            <Link
   397→              href="/admin/generer"
   398→              className="inline-flex items-center gap-2 px-4 py-2 bg-green-500 hover:bg-green-600 text-white rounded-xl font-medium transition-colors"
   399→            >
   400→              <QrCode className="w-4 h-4" />
   401→              Générer des QR codes
   402→            </Link>
   403→          )}
   404→        </div>
   405→      ) : (
   406→        <div className="space-y-4">
   407→          {agencyGroups.map((group) => (
   408→            <div
   409→              key={group.agencyId || 'no-agency'}
   410→              className="bg-white dark:bg-slate-800 rounded-2xl border border-slate-200 dark:border-slate-700 overflow-hidden"
   411→            >
   412→              {/* Agency Header */}
   413→              <button
   414→                onClick={() => toggleAgencyGroup(group.agencyId)}
   415→                className="w-full flex items-center justify-between p-4 hover:bg-slate-50 dark:hover:bg-slate-700/50 transition-colors"
   416→              >
   417→                <div className="flex items-center gap-3">
   418→                  <div className={`w-10 h-10 rounded-xl flex items-center justify-center ${
   419→                    activeTab === 'hajj' 
   420→                      ? 'bg-emerald-100 dark:bg-blue-600/20' 
   421→                      : 'bg-amber-100 dark:bg-blue-600/20'
   422→                  }`}>
   423→                    <Building2 className={`w-5 h-5 ${
   424→                      activeTab === 'hajj' ? 'text-blue-700' : 'text-amber-600'
   425→                    }`} />
   426→                  </div>
   427→                  <div className="text-left">
   428→                    <h3 className="font-semibold text-slate-800 dark:text-white">
   429→                      {group.agencyName}
   430→                    </h3>
   431→                    <p className="text-sm text-slate-500 dark:text-slate-400">
   432→                      {group.sets.length} set{group.sets.length > 1 ? 's' : ''} • {group.totalQr} QR codes
   433→                    </p>
   434→                  </div>
   435→                </div>
   436→                <ChevronDown className={`w-5 h-5 text-slate-400 transition-transform ${
   437→                  group.isExpanded ? 'rotate-180' : ''
   438→                }`} />
   439→              </button>
   440→
   441→              {/* Sets in this agency */}
   442→              {group.isExpanded && (
   443→                <div className="border-t border-slate-200 dark:border-slate-700">
   444→                  <div className="divide-y divide-slate-100 dark:divide-slate-700">
   445→                    {group.sets.map((set) => (
   446→                      <div
   447→                        key={set.id}
   448→                        className="flex items-center justify-between p-4 hover:bg-slate-50 dark:hover:bg-slate-700/50 transition-colors"
   449→                      >
   450→                        <div className="flex items-center gap-4">
   451→                          {/* QR Icon */}
   452→                          <div className={`w-12 h-12 rounded-xl flex items-center justify-center ${
   453→                            activeTab === 'hajj'
   454→                              ? 'bg-emerald-100 dark:bg-blue-600/20'
   455→                              : 'bg-amber-100 dark:bg-blue-600/20'
   456→                          }`}>
   457→                            <QrCode className={`w-6 h-6 ${
   458→                              activeTab === 'hajj' ? 'text-blue-700' : 'text-amber-600'
   459→                            }`} />
   460→                          </div>
   461→
   462→                          {/* Info */}
   463→                          <div>
   464→                            <div className="flex items-center gap-2">
   465→                              <h4 className="font-medium text-slate-800 dark:text-white">
   466→                                {set.setId}
   467→                              </h4>
   468→                              <span className={`px-2 py-0.5 rounded-full text-xs font-medium ${
   469→                                set.activationStatus === 'activated'
   470→                                  ? 'bg-green-100 text-green-700 dark:bg-green-500/20 dark:text-green-400'
   471→                                  : set.activationStatus === 'partial'
   472→                                  ? 'bg-yellow-100 text-yellow-700 dark:bg-yellow-500/20 dark:text-yellow-400'
   473→                                  : 'bg-slate-100 text-slate-600 dark:bg-slate-700 dark:text-slate-400'
   474→                              }`}>
   475→                                {set.activationStatus === 'activated' ? 'Activé' : set.activationStatus === 'partial' ? 'Partiel' : 'Nouveau'}
   476→                              </span>
   477→                            </div>
   478→                            <div className="flex items-center gap-4 mt-1 text-sm text-slate-500 dark:text-slate-400">
   479→                              <span>{set.qrCount} QR</span>
   480→                              {set.travelerName && (
   481→                                <span>• {set.travelerName}</span>
   482→                              )}
   483→                              <span>• {formatDate(set.createdAt)}</span>
   484→                            </div>
   485→                          </div>
   486→                        </div>
   487→
   488→                        {/* Actions */}
   489→                        <div className="flex items-center gap-2">
   490→                          <button
   491→                            onClick={() => {
   492→                              setSelectedSet(set);
   493→                              setShowDetailModal(true);
   494→                            }}
   495→                            className="p-2 text-slate-400 hover:text-blue-500 hover:bg-blue-50 dark:hover:bg-blue-500/10 rounded-lg transition-colors"
   496→                            title="Voir les QR codes"
   497→                          >
   498→                            <Eye className="w-4 h-4" />
   499→                          </button>
   500→                          <button
   501→                            onClick={() => handleDownloadPDF(set)}
   502→                            className="p-2 text-slate-400 hover:text-blue-600 hover:bg-emerald-50 dark:hover:bg-blue-600/10 rounded-lg transition-colors"
   503→                            title="Télécharger"
   504→                          >
   505→                            <Download className="w-4 h-4" />
   506→                          </button>
   507→                          <button
   508→                            onClick={() => handleShareSet(set)}
   509→                            className="p-2 text-slate-400 hover:text-purple-500 hover:bg-purple-50 dark:hover:bg-purple-500/10 rounded-lg transition-colors"
   510→                            title="Partager"
   511→                          >
   512→                            <Share2 className="w-4 h-4" />
   513→                          </button>
   514→                          <button
   515→                            onClick={() => {
   516→                              setSelectedSet(set);
   517→                              setShowDeleteModal(true);
   518→                            }}
   519→                            className="p-2 text-slate-400 hover:text-red-500 hover:bg-red-50 dark:hover:bg-red-500/10 rounded-lg transition-colors"
   520→                            title="Supprimer"
   521→                          >
   522→                            <Trash2 className="w-4 h-4" />
   523→                          </button>
   524→                        </div>
   525→                      </div>
   526→                    ))}
   527→                  </div>
   528→                </div>
   529→              )}
   530→            </div>
   531→          ))}
   532→        </div>
   533→      )}
   534→
   535→      {/* Detail Modal */}
   536→      {showDetailModal && selectedSet && (
   537→        <div className="fixed inset-0 bg-black/50 flex items-center justify-center p-4 z-50 backdrop-blur-sm">
   538→          <div className="bg-white dark:bg-slate-800 rounded-2xl max-w-2xl w-full max-h-[90vh] overflow-y-auto">

   542→                <p className="text-slate-500 dark:text-slate-400 text-sm">

   546→              </div>
   547→              <button
   548→                onClick={() => {

   550→                  setSelectedSet(null);

   554→                <X className="w-5 h-5 text-slate-400" />

   556→            </div>
   557→            <div className="p-6">

   559→              <div
   560→                ref={qrRef}
   561→                className="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 gap-4"
   562→              >
   563→                {selectedSet.references.map((ref, index) => (

   565→                    key={ref}
   566→                    className="bg-slate-50 dark:bg-slate-700/50 rounded-xl p-4 text-center"
   567→                  >
   568→                    <QRCodeSVG
   569→                      value={`${typeof window !== 'undefined' ? window.location.origin : ''}/found/${ref}`}

   571→                      level="H"
   572→                      includeMargin={true}
   573→                      bgColor="#f8fafc"
   574→                      fgColor={selectedSet.type === 'hajj' ? '#1D4ED8' : '#f59e0b'}

   576→                    <p className="text-slate-800 dark:text-white font-mono font-bold mt-2 text-sm">
   577→                      {ref}
   578→                    </p>
   579→                    <p className="text-slate-500 dark:text-slate-400 text-xs">
   580→                      {selectedSet.type === 'hajj' 
   581→                        ? (index === 0 ? 'Cabine' : `Soute #${index}`)
   582→                        : `Bagage #${index + 1}`
   583→                      }
   584→                    </p>
   585→                  </div>
   586→                ))}
   587→              </div>
   588→
   589→              {/* Info */}
   590→              <div className="mt-6 grid grid-cols-1 sm:grid-cols-2 gap-4">
   591→                <div className="bg-slate-50 dark:bg-slate-700/50 rounded-xl p-4">
   592→                  <p className="text-slate-500 dark:text-slate-400 text-sm">Créé le</p>

   594→                </div>
   595→                <div className="bg-slate-50 dark:bg-slate-700/50 rounded-xl p-4">
   596→                  <p className="text-slate-500 dark:text-slate-400 text-sm">Statut</p>
   597→                  <p className="text-slate-800 dark:text-white font-medium capitalize">
   598→                    {selectedSet.activationStatus === 'activated' ? 'Activé' : 
   599→                     selectedSet.activationStatus === 'partial' ? 'Partiel' : 'Nouveau'}
   600→                  </p>
   601→                </div>
   602→              </div>
   603→
   604→              {/* Actions */}
   605→              <div className="mt-6 flex gap-3">
   606→                <button
   607→                  onClick={() => handleDownloadPDF(selectedSet)}
   608→                  className="flex-1 py-3 bg-indigo-600 text-white rounded-xl hover:bg-indigo-700 transition-colors flex items-center justify-center gap-2"
   609→                >
   610→                  <FileText className="w-4 h-4" />
   611→                  Télécharger PDF
   612→                </button>
   613→                <button
   614→                  onClick={() => handleShareSet(selectedSet)}
   615→                  className="flex-1 py-3 bg-blue-700 text-white rounded-xl hover:bg-emerald-700 transition-colors flex items-center justify-center gap-2"
   616→                >
   617→                  <Share2 className="w-4 h-4" />
   618→                  Partager
   619→                </button>
   620→              </div>
   621→            </div>
   622→          </div>
   623→        </div>
   624→      )}
   625→
   626→      {/* Delete Confirmation Modal */}
   627→      {showDeleteModal && selectedSet && (
   628→        <div className="fixed inset-0 bg-black/50 flex items-center justify-center p-4 z-50 backdrop-blur-sm">
   629→          <div className="bg-white dark:bg-slate-800 rounded-2xl max-w-sm w-full">
   630→            <div className="p-6">
   631→              <div className="flex items-center gap-3 mb-4">
   632→                <div className="w-10 h-10 bg-red-100 dark:bg-red-900/30 rounded-xl flex items-center justify-center">
   633→                  <AlertTriangle className="w-5 h-5 text-red-600" />
   634→                </div>
   635→                <div>
   636→                  <h3 className="text-slate-800 dark:text-white font-bold">Supprimer ce set ?</h3>
   637→                  <p className="text-slate-500 dark:text-slate-400 text-sm">{selectedSet.setId}</p>
   638→                </div>
   639→              </div>
   640→              <p className="text-slate-500 dark:text-slate-400 text-sm mb-6">
   641→                Cette action supprimera définitivement les {selectedSet.qrCount} QR codes de ce set.
   642→              </p>
   643→              <div className="flex gap-3">
   644→                <button
   645→                  onClick={() => {
   646→                    setShowDeleteModal(false);
   647→                    setSelectedSet(null);
   648→                  }}
   649→                  className="flex-1 py-2 px-4 bg-slate-100 dark:bg-slate-700 text-slate-700 dark:text-slate-300 rounded-xl hover:bg-slate-200 dark:hover:bg-slate-600 transition-colors"
   650→                >
   651→                  Annuler
   652→                </button>
   653→                <button
   654→                  onClick={handleDeleteSet}
   655→                  className="flex-1 py-2 px-4 bg-red-600 text-white rounded-xl hover:bg-red-700 transition-colors"
   656→                >
   657→                  Supprimer
   658→                </button>
   659→              </div>
   660→            </div>
   661→          </div>
   662→        </div>
   663→      )}
   664→    </div>
