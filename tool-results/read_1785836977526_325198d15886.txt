   246→   246→// ─── Helper: Parse hotelCoords ───
   247→   247→function parseCoords(coords: { lat: number; lng: number } | string | null): { lat: number; lng: number } | null {
   248→   248→  if (!coords) return null;
   249→   249→  if (typeof coords === 'object' && 'lat' in coords && 'lng' in coords) {
   250→   250→    return { lat: coords.lat, lng: coords.lng };
   251→   251→  }
   252→   252→  if (typeof coords === 'string') {
   253→   253→    try {
   254→   254→      const parsed = JSON.parse(coords);
   255→   255→      if (parsed && typeof parsed === 'object' && 'lat' in parsed && 'lng' in parsed) {
   256→   256→        return { lat: parsed.lat, lng: parsed.lng };
   257→   257→      }
   258→   258→    } catch {
   259→   259→      const parts = coords.split(',').map((s) => parseFloat(s.trim()));
   260→   260→      if (parts.length === 2 && !isNaN(parts[0]) && !isNaN(parts[1])) {
   261→   261→        return { lat: parts[0], lng: parts[1] };
   262→   262→      }
   263→   263→    }
   264→   264→  }
   265→   265→  return null;
   266→   266→}
