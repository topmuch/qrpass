     1→'use client';
     2→
     3→import { useState, useEffect, useRef } from 'react';
     4→import Link from 'next/link';
     5→import { QRCodeSVG } from 'qrcode.react';
     6→import {
     7→  Search,
     8→  Eye,
     9→  Download,
    10→  Share2,
    11→  Trash2,
    12→  Plane,
    13→  Luggage,
    14→  QrCode,
    15→  X,
    16→  AlertTriangle,
    17→  RefreshCw,
    18→  FileText,
    19→  Building2,
    20→  ChevronDown,
    21→  ChevronRight,
    22→  UserRound,
    23→} from "lucide-react";
    24→
    25→interface QRSet {
    26→  id: string;
    27→  setId: string;
    28→  type: string;
    29→  agencyId: string | null;
    30→  agencyName: string | null;
    31→  createdAt: Date;
    32→  qrCount: number;
    33→  references: string[];
    34→  status: string;
    35→  travelerName: string | null;
    36→  activationStatus: 'new' | 'partial' | 'activated';
    37→  baggageIds: string[];
    38→}
    39→
    40→interface AgencyGroup {
    41→  agencyId: string | null;
    42→  agencyName: string;
    43→  sets: QRSet[];
    44→  totalQr: number;
    45→  isExpanded: boolean;
    46→}
    47→
    48→interface Stats {
    49→  totalSets: number;
    50→  totalQr: number;
    51→  hajjSets: number;
    52→}
    53→
    54→export default function EtiquettesPage() {
    55→  const qrRef = useRef<HTMLDivElement>(null);
    56→
    57→  const [sets, setSets] = useState<QRSet[]>([]);
    58→  const [agencyGroups, setAgencyGroups] = useState<AgencyGroup[]>([]);
    59→  const [stats, setStats] = useState<Stats>({
    60→    totalSets: 0,
    61→    totalQr: 0,
    62→    hajjSets: 0,
    63→
    64→  });
    65→  const [loading, setLoading] = useState(true);
    66→  const [search, setSearch] = useState('');
    67→  const [activeTab, setActiveTab] = useState<'hajj'>('hajj');
    68→
    69→  // Modals
    70→  const [showDetailModal, setShowDetailModal] = useState(false);
    71→  const [showDeleteModal, setShowDeleteModal] = useState(false);
    72→  const [selectedSet, setSelectedSet] = useState<QRSet | null>(null);
    73→  const [isDownloading, setIsDownloading] = useState(false);
    74→
    75→  useEffect(() => {
    76→    fetchSets();
    77→  }, [activeTab, search]);
    78→
    79→  const fetchSets = async () => {
    80→    setLoading(true);
    81→    try {
    82→      const params = new URLSearchParams();
    83→      params.set('type', activeTab);
    84→      if (search) params.set('search', search);
    85→
    86→      const response = await fetch(`/api/qrcodes?${params}`);
    87→      const data = await response.json();
    88→
    89→      // Calculate activation status for each set
    90→      const setsWithStatus = data.sets.map((set: QRSet) => ({
    91→        ...set,
    92→        activationStatus: getActivationStatus(set.status)
    93→      }));
    94→
    95→      // Filter to only show sets matching the active tab type
    96→      const filteredSets = setsWithStatus.filter((set: QRSet) => set.type === activeTab);
    97→      
    98→      setSets(filteredSets);
    99→      setStats(data.stats);
   100→
   101→      // Group by agency
   102→      const groupedByAgency = groupByAgency(filteredSets);
   103→      setAgencyGroups(groupedByAgency);
   104→    } catch (error) {
   105→      console.error('Error fetching QR sets:', error);
   106→    } finally {
   107→      setLoading(false);
   108→    }
   109→  };
   110→
   111→  const groupByAgency = (sets: QRSet[]): AgencyGroup[] => {
   112→    const groups = new Map<string | null, AgencyGroup>();
   113→
   114→    sets.forEach(set => {
   115→      const key = set.agencyId || 'no-agency';
   116→      const agencyName = set.agencyName || 'Sans agence';
   117→
   118→      if (!groups.has(key)) {
   119→        groups.set(key, {
   120→          agencyId: set.agencyId,
   121→          agencyName,
   122→          sets: [],
   123→          totalQr: 0,
   124→          isExpanded: true, // Expanded by default
   125→        });
   126→      }
   127→
   128→      const group = groups.get(key)!;
   129→      group.sets.push(set);
   130→      group.totalQr += set.qrCount;
   131→    });
   132→
   133→    // Sort: agencies with name first, then "Sans agence"
   134→    return Array.from(groups.values()).sort((a, b) => {
   135→      if (a.agencyId === null) return 1;
   136→      if (b.agencyId === null) return -1;
   137→      return a.agencyName.localeCompare(b.agencyName);
   138→    });
   139→  };
   140→
   141→  const getActivationStatus = (status: string): 'new' | 'partial' | 'activated' => {
   142→    if (status === 'active' || status === 'scanned') return 'activated';
   143→    if (status === 'partial') return 'partial';
   144→    return 'new';
   145→  };
   146→
   147→  const toggleAgencyGroup = (agencyId: string | null) => {
   148→    setAgencyGroups(prev => 
   149→      prev.map(group => 
   150→        group.agencyId === agencyId 
   151→          ? { ...group, isExpanded: !group.isExpanded }
   152→          : group
   153→      )
   154→    );
   155→  };
   156→
   157→  const handleDeleteSet = async () => {
   158→    if (!selectedSet) return;
   159→
   160→    try {
   161→      const params = new URLSearchParams({ setId: selectedSet.setId });
   162→      const response = await fetch(`/api/qrcodes?${params}`, { method: 'DELETE' });
   163→      const data = await response.json();
   164→
   165→      if (response.ok && data.success) {
   166→        fetchSets();
   167→        setShowDeleteModal(false);
   168→        setSelectedSet(null);
   169→      } else {
   170→        alert(`Erreur: ${data.error || 'Erreur lors de la suppression'}`);
   171→      }
   172→    } catch (error) {
   173→      console.error('Error deleting set:', error);
   174→      alert('Erreur lors de la suppression');
   175→    }
   176→  };
   177→
   178→  const handleDownloadPDF = async (set: QRSet) => {
   179→    setIsDownloading(true);
   180→    setSelectedSet(set);
   181→
   182→    try {
   183→      const canvas = document.createElement('canvas');
   184→      const ctx = canvas.getContext('2d');
   185→      if (!ctx) throw new Error('Canvas not supported');
   186→
   187→      const qrSize = 300;
   188→      const padding = 40;
   189→      const headerHeight = 100;
   190→
   191→      const cols = Math.min(set.qrCount, 3);
   192→      const rows = Math.ceil(set.qrCount / 3);
   193→
   194→      canvas.width = cols * qrSize + (cols + 1) * padding;
   195→      canvas.height = headerHeight + rows * qrSize + (rows + 1) * padding + 60;
   196→
   197→      // Background
   198→      ctx.fillStyle = '#ffffff';
   199→      ctx.fillRect(0, 0, canvas.width, canvas.height);
   200→
   201→      // Header
   202→      ctx.fillStyle = set.type === 'hajj' ? '#1D4ED8' : '#f59e0b';
   203→      ctx.fillRect(0, 0, canvas.width, headerHeight);
   204→
   205→      ctx.fillStyle = '#ffffff';
   206→      ctx.font = 'bold 24px Arial';
   207→      ctx.textAlign = 'center';
   208→      ctx.fillText('PassHajj - Étiquettes', canvas.width / 2, 35);
   209→
   210→      ctx.font = '16px Arial';
   211→      ctx.fillText(`${set.setId} | Hajj 2026 | ${set.qrCount} QR`, canvas.width / 2, 65);
   212→      if (set.agencyName) {
   213→        ctx.font = '14px Arial';
   214→        ctx.fillText(`Agence: ${set.agencyName}`, canvas.width / 2, 85);
   215→      }
   216→
   217→      // Generate QR codes
   218→      for (let i = 0; i < set.references.length; i++) {
   219→        const col = i % 3;
   220→        const row = Math.floor(i / 3);
   221→        const x = padding + col * (qrSize + padding);
   222→        const y = headerHeight + padding + row * (qrSize + padding);
   223→
   224→        ctx.fillStyle = '#ffffff';
   225→        ctx.fillRect(x, y, qrSize, qrSize);
   226→
   227→        const qrUrl = `${window.location.origin}/found/${set.references[i]}`;
   228→        const svgElement = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
   229→        const qrSvg = <QRCodeSVG value={qrUrl} size={qrSize - 40} level="H" fgColor={set.type === 'hajj' ? '#1D4ED8' : '#f59e0b'} />;
   230→
   231→        // Use a promise-based approach to render SVG to canvas
   232→        const svgString = `<svg xmlns="http://www.w3.org/2000/svg" width="${qrSize - 40}" height="${qrSize - 40}">
   233→          <rect width="100%" height="100%" fill="white"/>
   234→          ${generateQRPath(qrUrl, qrSize - 40, set.type === 'hajj' ? '#1D4ED8' : '#f59e0b')}
   235→        </svg>`;
   236→
   237→        const img = new Image();
   238→        const svgBlob = new Blob([svgString], { type: 'image/svg+xml;charset=utf-8' });
   239→        const url = URL.createObjectURL(svgBlob);
   240→
   241→        await new Promise<void>((resolve) => {
   242→          img.onload = () => {
   243→            ctx.drawImage(img, x + 20, y + 20, qrSize - 40, qrSize - 40);
   244→            URL.revokeObjectURL(url);
   245→            resolve();
   246→          };
   247→          img.onerror = () => {
   248→            URL.revokeObjectURL(url);
   249→            resolve();
   250→          };
   251→          img.src = url;
   252→        });
   253→
   254→        // Reference label
   255→        ctx.fillStyle = '#1e293b';
   256→        ctx.font = 'bold 12px Arial';
   257→        ctx.textAlign = 'center';
   258→        ctx.fillText(set.references[i], x + qrSize / 2, y + qrSize - 15);
   259→      }
   260→
   261→      // Footer
   262→      ctx.fillStyle = '#64748b';
   263→      ctx.font = '12px Arial';
   264→      ctx.textAlign = 'center';
   265→      ctx.fillText(`Généré le ${new Date().toLocaleDateString('fr-FR')} | QRPass.com`, canvas.width / 2, canvas.height - 20);
   266→
   267→      // Download
   268→      const link = document.createElement('a');
   269→      link.download = `PassHajj-${set.setId}.png`;
   270→      link.href = canvas.toDataURL('image/png');
   271→      link.click();
   272→
   273→      setIsDownloading(false);
   274→    } catch (error) {
   275→      console.error('Error generating PDF:', error);
   276→      setIsDownloading(false);
   277→      alert('Erreur lors de la génération');
   278→    }
   279→  };
   280→
   281→  // Simple QR path generator fallback
   282→  const generateQRPath = (url: string, size: number, color: string): string => {
   283→    return `<text x="50%" y="50%" text-anchor="middle" fill="${color}" font-size="10">${url}</text>`;
   284→  };
   285→
   286→  const handleShareSet = async (set: QRSet) => {
   287→    const shareUrl = `${window.location.origin}/found/${set.references[0]}`;
   288→    if (navigator.share) {
   289→      try {
   290→        await navigator.share({
   291→          title: `PassHajj - ${set.setId}`,
   292→          text: `${set.qrCount} QR codes pour ${set.agencyName || 'agence'}`,
   293→          url: shareUrl,
   294→        });
   295→      } catch (err) {
   296→        console.log('Share cancelled');
   297→      }
   298→    } else {
   299→      navigator.clipboard.writeText(shareUrl);
   300→      alert('Lien copié !');
   301→    }
   302→  };
   303→
   304→  const formatDate = (date: Date | string) => {
   305→    return new Date(date).toLocaleDateString('fr-FR', {
   306→      day: '2-digit',
   307→      month: 'short',
   308→      year: 'numeric',
   309→    });
   310→  };
   311→
   312→  return (
   313→    <div className="max-w-7xl mx-auto">
   314→      {/* Header */}
   315→      <div className="flex flex-col md:flex-row md:items-center justify-between gap-4 mb-6">
   316→        <div>
   317→          <h1 className="text-2xl font-bold text-slate-800 dark:text-white">Étiquettes QR</h1>
   318→          <p className="text-slate-500 dark:text-slate-400 text-sm">
   319→            {stats.totalSets} sets • {stats.totalQr} QR codes
   320→          </p>
   321→        </div>
   322→
   323→        <div className="flex items-center gap-3">
   324→          <div className="relative">
   325→            <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" />
   326→            <input
   327→              type="text"
   328→              placeholder="Rechercher..."
   329→              value={search}
   330→              onChange={(e) => setSearch(e.target.value)}
   331→              className="pl-10 pr-4 py-2 w-64 bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-xl text-sm focus:outline-none focus:ring-2 focus:ring-green-500"
   332→            />
   333→          </div>
   334→
   335→          <button
   336→            onClick={() => fetchSets()}
   337→            className="p-2 bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-xl hover:bg-slate-50 dark:hover:bg-slate-700 transition-colors"
   338→          >
   339→            <RefreshCw className={`w-4 h-4 text-slate-500 ${loading ? 'animate-spin' : ''}`} />
   340→          </button>
   341→
   342→          <Link
   343→            href="/admin/generer"
   344→            className="flex items-center gap-2 px-4 py-2 bg-green-500 hover:bg-green-600 text-white rounded-xl font-medium transition-colors"
   345→          >
   346→            <QrCode className="w-4 h-4" />
   347→            Générer QR
   348→          </Link>
   349→        </div>
   350→      </div>
   351→
   352→      {/* Tabs */}
   353→      <div className="flex gap-2 mb-6">
   354→        <button
   355→          onClick={() => setActiveTab('hajj')}
   356→          className={`flex items-center gap-2 px-6 py-3 rounded-xl font-medium transition-all ${
   357→            activeTab === 'hajj'
   358→              ? 'bg-blue-700 text-white shadow-lg shadow-blue-700/30'
   359→              : 'bg-white dark:bg-slate-800 text-slate-600 dark:text-slate-300 border border-slate-200 dark:border-slate-700 hover:bg-slate-50 dark:hover:bg-slate-700'
   360→          }`}
   361→        >
   362→          <Luggage className="w-5 h-5" />
   363→          Hajj
   364→          <span className={`px-2 py-0.5 rounded-full text-xs ${
   365→            activeTab === 'hajj' 
   366→              ? 'bg-white/20 text-white' 
   367→              : 'bg-slate-100 dark:bg-slate-700 text-slate-600 dark:text-slate-400'
   368→          }`}>
   369→            {stats.hajjSets}
   370→          </span>
   371→        </button>
   372→      </div>
   373→
   374→      {/* Content */}
   375→      {loading ? (
   376→        <div className="flex items-center justify-center py-20">
   377→          <div className="animate-spin w-8 h-8 border-4 border-slate-200 border-t-green-500 rounded-full" />
   378→        </div>
   379→      ) : agencyGroups.length === 0 ? (
   380→        <div className="bg-white dark:bg-slate-800 rounded-2xl border border-slate-200 dark:border-slate-700 p-12 text-center">
   381→          <div className={`w-16 h-16 mx-auto mb-4 rounded-2xl flex items-center justify-center ${
   382→            activeTab === 'hajj' ? 'bg-emerald-100 dark:bg-blue-600/20' : 'bg-amber-100 dark:bg-blue-600/20'
   383→          }`}>
   384→            {activeTab === 'hajj' ? (
   385→              <Luggage className="w-8 h-8 text-blue-700" />
   386→            ) : (
   387→              <Plane className="w-8 h-8 text-amber-600" />
   388→            )}
   389→          </div>
   390→          <h3 className="text-lg font-semibold text-slate-800 dark:text-white mb-2">
   391→            Aucun QR code Hajj
   392→          </h3>
   393→          <p className="text-slate-500 dark:text-slate-400 mb-4">
   394→            {search ? 'Aucun résultat pour votre recherche' : 'Commencez par générer des QR codes'}
   395→          </p>
   396→          {!search && (
   397→            <Link
   398→              href="/admin/generer"
   399→              className="inline-flex items-center gap-2 px-4 py-2 bg-green-500 hover:bg-green-600 text-white rounded-xl font-medium transition-colors"
   400→            >
   401→              <QrCode className="w-4 h-4" />
   402→              Générer des QR codes
   403→            </Link>
   404→          )}
   405→        </div>
   406→      ) : (
   407→        <div className="space-y-4">
   408→          {agencyGroups.map((group) => (
   409→            <div
   410→              key={group.agencyId || 'no-agency'}
   411→              className="bg-white dark:bg-slate-800 rounded-2xl border border-slate-200 dark:border-slate-700 overflow-hidden"
   412→            >
   413→              {/* Agency Header */}
   414→              <button
   415→                onClick={() => toggleAgencyGroup(group.agencyId)}
   416→                className="w-full flex items-center justify-between p-4 hover:bg-slate-50 dark:hover:bg-slate-700/50 transition-colors"
   417→              >
   418→                <div className="flex items-center gap-3">
   419→                  <div className={`w-10 h-10 rounded-xl flex items-center justify-center ${
   420→                    activeTab === 'hajj' 
   421→                      ? 'bg-emerald-100 dark:bg-blue-600/20' 
   422→                      : 'bg-amber-100 dark:bg-blue-600/20'
   423→                  }`}>
   424→                    <Building2 className={`w-5 h-5 ${
   425→                      activeTab === 'hajj' ? 'text-blue-700' : 'text-amber-600'
   426→                    }`} />
   427→                  </div>
   428→                  <div className="text-left">
   429→                    <h3 className="font-semibold text-slate-800 dark:text-white">
   430→                      {group.agencyName}
   431→                    </h3>
   432→                    <p className="text-sm text-slate-500 dark:text-slate-400">
   433→                      {group.sets.length} set{group.sets.length > 1 ? 's' : ''} • {group.totalQr} QR codes
   434→                    </p>
   435→                  </div>
   436→                </div>
   437→                <ChevronDown className={`w-5 h-5 text-slate-400 transition-transform ${
   438→                  group.isExpanded ? 'rotate-180' : ''
   439→                }`} />
   440→              </button>
   441→
   442→              {/* Sets in this agency */}
   443→              {group.isExpanded && (
   444→                <div className="border-t border-slate-200 dark:border-slate-700">
   445→                  <div className="divide-y divide-slate-100 dark:divide-slate-700">
   446→                    {group.sets.map((set) => (
   447→                      <div
   448→                        key={set.id}
   449→                        className="flex items-center justify-between p-4 hover:bg-slate-50 dark:hover:bg-slate-700/50 transition-colors"
   450→                      >
   451→                        <div className="flex items-center gap-4">
   452→                          {/* QR Icon with type badge */}
   453→                          <div className={`w-12 h-12 rounded-xl flex items-center justify-center ${
   454→                            set.references[0]?.startsWith('PH-P-')
   455→                              ? 'bg-blue-100 dark:bg-blue-600/20'
   456→                              : activeTab === 'hajj'
   457→                                ? 'bg-amber-100 dark:bg-amber-600/20'
   458→                                : 'bg-amber-100 dark:bg-blue-600/20'
   459→                          }`}>
   460→                            {set.references[0]?.startsWith('PH-P-') ? (
   461→                              <UserRound className="w-6 h-6 text-blue-600" />
   462→                            ) : (
   463→                              <Luggage className="w-6 h-6 text-amber-600" />
   464→                            )}
   465→                          </div>
   466→
   467→                          {/* Info */}
   468→                          <div>
   469→                            <div className="flex items-center gap-2">
   470→                              <h4 className="font-medium text-slate-800 dark:text-white">
   471→                                {set.setId}
   472→                              </h4>
   473→                              {/* Type Badge */}
   474→                              {set.references[0]?.startsWith('PH-P-') ? (
   475→                                <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-xs font-medium bg-blue-100 text-blue-700 dark:bg-blue-500/20 dark:text-blue-400">
   476→                                  <UserRound className="w-3 h-3" /> Identity
   477→                                </span>
   478→                              ) : (
   479→                                <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-xs font-medium bg-amber-100 text-amber-700 dark:bg-amber-500/20 dark:text-amber-400">
   480→                                  <Luggage className="w-3 h-3" /> Bagage
   481→                                </span>
   482→                              )}
   483→                              <span className={`px-2 py-0.5 rounded-full text-xs font-medium ${
   484→                                set.activationStatus === 'activated'
   485→                                  ? 'bg-green-100 text-green-700 dark:bg-green-500/20 dark:text-green-400'
   486→                                  : set.activationStatus === 'partial'
   487→                                  ? 'bg-yellow-100 text-yellow-700 dark:bg-yellow-500/20 dark:text-yellow-400'
   488→                                  : 'bg-slate-100 text-slate-600 dark:bg-slate-700 dark:text-slate-400'
   489→                              }`}>
   490→                                {set.activationStatus === 'activated' ? 'Activé' : set.activationStatus === 'partial' ? 'Partiel' : 'Nouveau'}
   491→                              </span>
   492→                            </div>
   493→                            <div className="flex items-center gap-4 mt-1 text-sm text-slate-500 dark:text-slate-400">
   494→                              <span>{set.qrCount} QR</span>
   495→                              {set.travelerName && (
   496→                                <span>• {set.travelerName}</span>
   497→                              )}
   498→                              <span>• {formatDate(set.createdAt)}</span>
   499→                            </div>
   500→                          </div>
   501→                        </div>
   502→
   503→                        {/* Actions */}
   504→                        <div className="flex items-center gap-2">
   505→                          <button
   506→                            onClick={() => {
   507→                              setSelectedSet(set);
   508→                              setShowDetailModal(true);
   509→                            }}
   510→                            className="p-2 text-slate-400 hover:text-blue-500 hover:bg-blue-50 dark:hover:bg-blue-500/10 rounded-lg transition-colors"
   511→                            title="Voir les QR codes"
   512→                          >
   513→                            <Eye className="w-4 h-4" />
   514→                          </button>
   515→                          <button
   516→                            onClick={() => handleDownloadPDF(set)}
   517→                            className="p-2 text-slate-400 hover:text-blue-600 hover:bg-emerald-50 dark:hover:bg-blue-600/10 rounded-lg transition-colors"
   518→                            title="Télécharger"
   519→                          >
   520→                            <Download className="w-4 h-4" />
   521→                          </button>
   522→                          <button
   523→                            onClick={() => handleShareSet(set)}

   642→      {showDeleteModal && selectedSet && (
