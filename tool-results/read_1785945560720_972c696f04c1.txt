    67→// ─── Status helpers ───
    68→type TripStatus = 'active' | 'completed' | 'cancelled' | 'pending';
    69→
    70→function getStatusBadge(status: string) {
    71→  const s = status?.toLowerCase() as TripStatus;
    72→  switch (s) {
    73→    case 'active':
    74→      return (
    75→        <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200 hover:bg-emerald-100">
    76→          Actif
    77→        </Badge>
    78→      );
    79→    case 'completed':
    80→      return (
    81→        <Badge className="bg-blue-100 text-blue-700 border-blue-200 hover:bg-blue-100">
    82→          Terminé
    83→        </Badge>
    84→      );
    85→    case 'cancelled':
    86→      return (
    87→        <Badge className="bg-red-100 text-red-700 border-red-200 hover:bg-red-100">
    88→          Annulé
    89→        </Badge>
    90→      );
    91→    case 'pending':
    92→      return (
    93→        <Badge className="bg-amber-100 text-amber-700 border-amber-200 hover:bg-amber-100">
    94→          En attente
    95→        </Badge>
    96→      );
    97→    default:
    98→      return <Badge variant="secondary">{status}</Badge>;
    99→  }
   100→}
   101→
   102→function getTransportIcon(mode: string | null | undefined) {
   103→  switch (mode?.toLowerCase()) {
   104→    case 'flight':
   105→    case 'avion':

   113→      return <Train className="w-4 h-4" />;
   114→    default:

   116→  }
   117→}

   119→function formatDate(dateStr: string | null | undefined): string {
   120→  if (!dateStr) return '—';
   121→  try {
   122→    const d = new Date(dateStr);
   123→    return d.toLocaleDateString('fr-FR', {
   124→      day: '2-digit',
   125→      month: 'short',
   126→      year: 'numeric',
   127→    });
   128→  } catch {
   129→    return '—';
   130→  }
   131→}
