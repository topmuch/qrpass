   246→// ─── Helper: Parse hotelCoords ───
   247→function parseCoords(coords: { lat: number; lng: number } | string | null): { lat: number; lng: number } | null {
   248→  if (!coords) return null;
   249→  if (typeof coords === 'object' && 'lat' in coords && 'lng' in coords) {
   250→    return { lat: coords.lat, lng: coords.lng };
   251→  }
   252→  if (typeof coords === 'string') {
   253→    try {
   254→      const parsed = JSON.parse(coords);
   255→      if (parsed && typeof parsed === 'object' && 'lat' in parsed && 'lng' in parsed) {
   256→        return { lat: parsed.lat, lng: parsed.lng };
   257→      }
   258→    } catch {
   259→      const parts = coords.split(',').map((s) => parseFloat(s.trim()));
   260→      if (parts.length === 2 && !isNaN(parts[0]) && !isNaN(parts[1])) {
   261→        return { lat: parts[0], lng: parts[1] };
   262→      }
   263→    }
   264→  }
   265→  return null;
   266→}
