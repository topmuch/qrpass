   217→// ─── Main Component ───
   218→function BaggageActivateContent() {
   219→  const router = useRouter();
   220→  const searchParams = useSearchParams();
   221→  const qrFromUrl = searchParams.get('code') || searchParams.get('qr') || '';
   222→
   223→  // Language
   224→  const [lang, setLang] = useState<Lang>('fr');
   225→  const dir = lang === 'ar' ? 'rtl' : 'ltr';
   226→  const t = T[lang];
   227→
   228→  // Form state
   229→  const [reference, setReference] = useState('');
   230→  const [fullName, setFullName] = useState('');
   231→  const [countryCode, setCountryCode] = useState('+221');
   232→  const [phone, setPhone] = useState('');
   233→  const [email, setEmail] = useState('');
   234→  const [airline, setAirline] = useState('');
   235→  const [flightNumber, setFlightNumber] = useState('');
   236→  const [destination, setDestination] = useState('');
   237→  const [departureDate, setDepartureDate] = useState('');
   238→  const [departureTime, setDepartureTime] = useState('');
   239→  const [photoPreview, setPhotoPreview] = useState<string | null>(null);
   240→  const [photoFile, setPhotoFile] = useState<File | null>(null);
   241→  const [loading, setLoading] = useState(false);
   242→
   243→  // Deferred activation
   244→  const [deferredActivation, setDeferredActivation] = useState(false);
   245→  const [activationDate, setActivationDate] = useState('');
   246→
   247→  // Autocomplete state
   248→  const [airlineResults, setAirlineResults] = useState<string[]>([]);
   249→  const [showAirlineResults, setShowAirlineResults] = useState(false);
   250→  const airlineRef = useRef<HTMLDivElement>(null);
   251→
   252→  // Country code dropdown
   253→  const [showCountryDropdown, setShowCountryDropdown] = useState(false);
   254→  const countryRef = useRef<HTMLDivElement>(null);
   255→
   256→  // Pre-fill QR from URL
   257→  useEffect(() => {
   258→    if (qrFromUrl) {
   259→      setReference(qrFromUrl.toUpperCase());
   260→    }
   261→  }, [qrFromUrl]);
   262→
   263→  // Detect country by IP
   264→  useEffect(() => {
   265→    async function detectCountry() {
   266→      try {
   267→        const res = await fetch('/api/detect-country');
   268→        if (res.ok) {
   269→          const data = await res.json();
   270→          const dial = COUNTRY_TO_DIAL[data.countryCode];
   271→          if (dial) {
   272→            setCountryCode(dial);
   273→          }
   274→        }
   275→      } catch {
   276→        // Silently fail, keep default +221
   277→      }
   278→    }
   279→    detectCountry();
   280→  }, []);
   281→
   282→  // Close dropdowns on outside click
   283→  useEffect(() => {
   284→    function handleClick(e: MouseEvent) {
   285→      if (airlineRef.current && !airlineRef.current.contains(e.target as Node)) {
   286→        setShowAirlineResults(false);
   287→      }
   288→      if (countryRef.current && !countryRef.current.contains(e.target as Node)) {
   289→        setShowCountryDropdown(false);
   290→      }
   291→    }
   292→    document.addEventListener('mousedown', handleClick);
   293→    return () => document.removeEventListener('mousedown', handleClick);
   294→  }, []);
   295→
   296→  // Airline autocomplete handler
   297→  const handleAirlineInput = useCallback((value: string) => {
   298→    setAirline(value);
   299→    if (value.length < 2) {
   300→      setAirlineResults([]);
   301→      setShowAirlineResults(false);
   302→      return;
   303→    }
   304→    const filtered = AIRLINES.filter(a => a.toLowerCase().includes(value.toLowerCase()));
   305→    setAirlineResults(filtered);
   306→    setShowAirlineResults(filtered.length > 0);
   307→  }, []);
   308→
   309→  const selectAirline = useCallback((name: string) => {
   310→    setAirline(name);
   311→    setShowAirlineResults(false);
   312→  }, []);
   313→
   314→  // Photo preview
   315→  const handlePhotoChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   316→    const file = e.target.files?.[0];
   317→    if (file) {
   318→      setPhotoFile(file);
   319→      const reader = new FileReader();
   320→      reader.onload = (ev) => setPhotoPreview(ev.target?.result as string);
   321→      reader.readAsDataURL(file);
   322→    }
   323→  };
   324→
   325→  // Form validation
   326→  const validate = (): boolean => {
   327→    if (!reference.trim()) {
   328→      toast({ title: 'Code QR requis', variant: 'destructive' });
   329→      return false;
   330→    }
   331→    if (!fullName.trim()) {
   332→      toast({ title: 'Nom complet requis', variant: 'destructive' });
   333→      return false;
   334→    }
   335→    if (!phone.trim()) {
   336→      toast({ title: 'Numéro WhatsApp requis', variant: 'destructive' });
   337→      return false;
   338→    }
   339→    if (!airline.trim()) {
   340→      toast({ title: 'Compagnie aérienne requise', variant: 'destructive' });
   341→      return false;
   342→    }
   343→    if (!flightNumber.trim()) {
   344→      toast({ title: 'Numéro de vol requis', variant: 'destructive' });
   345→      return false;
   346→    }
   347→    if (!destination) {
   348→      toast({ title: 'Destination requise', variant: 'destructive' });
   349→      return false;
   350→    }
   351→    if (!departureDate) {
   352→      toast({ title: 'Date de départ requise', variant: 'destructive' });
   353→      return false;
   354→    }
   355→    return true;
   356→  };
   357→
   358→  // Submit
   359→  const handleSubmit = async (e: React.FormEvent) => {
   360→    e.preventDefault();
   361→    if (!validate()) return;
   362→
   363→    setLoading(true);
   364→    try {
   365→      // Split fullName into firstName and lastName for API compatibility
   366→      const nameParts = fullName.trim().split(/\s+/);
   367→      const firstName = nameParts[0] || '';
   368→      const lastName = nameParts.slice(1).join(' ') || firstName;
   369→
   370→      // Upload photo first if selected
   371→      let photoUrl: string | null = null;
   372→      if (photoFile) {
   373→        try {
   374→          const photoFormData = new FormData();
   375→          photoFormData.append('file', photoFile);
   376→          const uploadRes = await fetch('/api/baggage/upload-photo', {
   377→            method: 'POST',
   378→            body: photoFormData,
   379→          });
   380→          if (uploadRes.ok) {
   381→            const uploadData = await uploadRes.json();
   382→            photoUrl = uploadData.photoUrl;
   383→          }
   384→        } catch {
   385→          // Non-blocking, continue without photo
   386→        }
   387→      }
   388→
   389→      const response = await fetch('/api/activate', {
   390→        method: 'POST',
   391→        headers: { 'Content-Type': 'application/json' },
   392→        body: JSON.stringify({
   393→          reference: reference.toUpperCase(),
   394→          travelerFirstName: firstName,
   395→          travelerLastName: lastName,
   396→          whatsappOwner: `${countryCode}${phone.trim()}`,
   397→          airlineName: airline.trim(),
   398→          flightNumber: flightNumber.trim().toUpperCase(),
   399→          destination,
   400→          departureDate,
   401→          departureTime: departureTime || undefined,
   402→          transportMode: 'flight',
   403→          photoUrl: photoUrl || undefined,
   404→          activationDate: deferredActivation && activationDate ? activationDate : undefined,
   405→        }),
   406→      });
   407→
   408→      if (response.ok) {
   409→        const data = await response.json();
   410→        // Store activation data for confirmation page
   411→        sessionStorage.setItem('activationData', JSON.stringify({
   412→          reference: reference.toUpperCase(),
   413→          firstName,
   414→          lastName,
   415→          whatsapp: `${countryCode}${phone.trim()}`,
   416→          airlineName: airline,
   417→          flightNumber,
   418→          destination,
   419→          type: 'hajj',
   420→          activatedAt: new Date().toISOString(),
   421→          expiresAt: data.baggage?.expiresAt,
   422→          photoUrl: photoUrl || data.baggage?.photoUrl || null,
   423→        }));
   424→
   425→        const params = new URLSearchParams({
   426→          code: reference.toUpperCase(),
   427→          firstName,
   428→          lastName,
   429→          flight: flightNumber,
   430→          destination,
   431→          chefPhone: `${countryCode}${phone.trim()}`,
   432→        });
   433→        if (photoUrl) params.set('photo', photoUrl);
   434→        router.push(`/activate/confirmation?${params.toString()}`);
   435→      } else {
   436→        const error = await response.json();
   437→        toast({ title: error.message || t.errorTitle, variant: 'destructive' });
   438→      }
   439→    } catch {
   440→      toast({ title: t.errorMsg, variant: 'destructive' });
   441→    } finally {
   442→      setLoading(false);
   443→    }
   444→  };
   445→
   446→  // Min date = today
   447→  const today = new Date().toISOString().split('T')[0];
   448→
   449→  return (
   450→    <main dir={dir} className="min-h-screen flex flex-col items-center" style={{ background: BG, color: TEXT, padding: '16px' }}>
   451→
   452→      {/* ─── Header ─── */}
   453→      <div className="w-full max-w-[480px] flex items-center justify-between mb-6 animate-[fadeInDown_0.5s_ease]">
   454→        <div>
   455→          <div className="text-[28px] font-extrabold tracking-tight text-black">
   456→            <span className="text-white bg-black px-2 py-0.5 rounded-md mr-1">Pass</span>Hajj
   457→          </div>
   458→          <div className="text-sm mt-1 text-black/60">{t.subtitle}</div>
   459→        </div>
   460→        <LangSelector lang={lang} setLang={setLang} />
   461→      </div>
   462→
   463→      {/* ─── Progress Bar ─── */}
   464→      <div className="w-full max-w-[480px] mb-5">
   465→        <div className="h-1 bg-white/40 rounded-full overflow-hidden">
   466→          <div className="h-full bg-black rounded-full" style={{ width: '50%', transition: 'width 0.3s ease' }} />
   467→        </div>
   468→      </div>
   469→
   470→      {/* ─── Form Card ─── */}
   471→      <form onSubmit={handleSubmit} className="w-full max-w-[480px]">
   472→        <div
   473→          className="rounded-[20px] p-7 shadow-[0_8px_24px_rgba(0,0,0,0.12)] mb-4 animate-[fadeInUp_0.5s_ease_0.2s_both]"
   474→          style={{ background: CARD_BG }}
   475→        >
   476→          <div className="text-lg font-bold mb-5 flex items-center gap-2">
   477→            {t.title}
   478→          </div>
   479→
   480→          {/* QR Code (Auto-détecté) */}
   481→          <div className="mb-5">
   482→            <Label className="text-[13px] font-semibold mb-1.5 block">{t.qrCode}</Label>
   483→            <Input
   484→              value={reference}
   485→              onChange={(e) => setReference(e.target.value.toUpperCase())}
   486→              readOnly={!!qrFromUrl}
   487→              placeholder={t.qrPlaceholder}
   488→              className={`h-[52px] rounded-xl text-[15px] ${qrFromUrl ? 'bg-[#f1f5f9] cursor-not-allowed text-gray-600' : 'bg-[#f8fafc]'} border-[#e2e8f0]`}
   489→            />
   490→            {qrFromUrl && (
   491→              <p className="text-xs mt-1.5 text-gray-500">{t.qrDetected}</p>
   492→            )}
   493→          </div>
   494→
