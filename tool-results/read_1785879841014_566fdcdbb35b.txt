   379→   379→  useEffect(() => {

   382→   382→        const res = await fetch('/api/advertisements');
   383→   383→        const data = await res.json();
   384→   384→        if (data.advertisements && data.advertisements.length > 0) {
   385→   385→          setAds(data.advertisements);
   386→   386→        }
   387→   387→      } catch (err) {
   388→   388→        console.error('Error fetching ads:', err);

   390→   390→        setLoading(false);
   391→   391→      }
   392→   392→    };

   403→   403→
   404→   404→  if (loading || ads.length === 0) return null;
   405→   405→
   406→   406→  const ad = ads[currentIndex];
   407→   407→
   408→   408→  return (

   414→   414→            <div className="absolute bottom-0 left-0 right-0 bg-gradient-to-t from-black/60 to-transparent p-4">
   415→   415→              <p className="text-white text-sm font-medium">{ad.title}</p>
   416→   416→            </div>
   417→   417→          )}
   418→   418→        </a>
   419→   419→      ) : (
   420→   420→        <a href={ad.linkUrl || '#'} target="_blank" rel="noopener noreferrer" className="block p-6 bg-gradient-to-r from-amber-50 to-blue-50 dark:from-blue-600/10 dark:to-blue-500/10 text-center">
   421→   421→          <p className="text-slate-800 dark:text-white font-semibold">{ad.title}</p>
   422→   422→        </a>
   423→   423→      )}
   424→   424→      {ads.length > 1 && (
   425→   425→        <div className="absolute bottom-2 right-2 flex gap-1">
   426→   426→          {ads.map((_, i) => (
   427→   427→            <button key={i} onClick={() => setCurrentIndex(i)} className={`w-2 h-2 rounded-full transition-all ${i === currentIndex ? 'bg-white w-4' : 'bg-white/50'}`} />

   429→   429→        </div>
