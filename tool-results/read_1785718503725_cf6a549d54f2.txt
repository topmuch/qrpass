    51→/* ──────────────────────────────────────────────
    52→   Animated Counter
    53→   ────────────────────────────────────────────── */
    54→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
    55→  const [count, setCount] = useState(0);
    56→  const ref = useRef<HTMLSpanElement>(null);
    57→  const inView = useInView(ref, { once: true });
    58→
    59→  useEffect(() => {
    60→    if (!inView) return;
    61→    let start = 0;
    62→    const duration = 2000;
    63→    const step = target / (duration / 16);
    64→    const timer = setInterval(() => {
    65→      start += step;
    66→      if (start >= target) {
    67→        setCount(target);
    68→        clearInterval(timer);
    69→      } else {
    70→        setCount(Math.floor(start));
    71→      }
    72→    }, 16);
    73→    return () => clearInterval(timer);
    74→  }, [inView, target]);
    75→
    76→  return <span ref={ref}>{count.toLocaleString()}{suffix}</span>;
    77→}

    82→function FadeIn({
    83→  children,
    84→  className,
    85→  delay = 0,
    86→  direction = 'up',
    87→}: {
    88→  children: React.ReactNode;
    89→  className?: string;
    90→  delay?: number;
    91→  direction?: 'up' | 'down' | 'left' | 'right';
    92→}) {
    93→  const ref = useRef<HTMLDivElement>(null);
    94→  const inView = useInView(ref, { once: true, margin: '-60px' });
    95→  const directions = {
    96→    up: { y: 50, x: 0 },
    97→    down: { y: -50, x: 0 },
    98→    left: { x: 50, y: 0 },
    99→    right: { x: -50, y: 0 },
   100→  };
   101→  return (
   102→    <motion.div
   103→      ref={ref}
   104→      initial={{ opacity: 0, ...directions[direction] }}
   105→      animate={inView ? { opacity: 1, x: 0, y: 0 } : { opacity: 0, ...directions[direction] }}
   106→      transition={{ duration: 0.8, ease: [0.25, 0.46, 0.45, 0.94], delay }}
   107→      className={className}
   108→    >
   109→      {children}
   110→    </motion.div>
   111→  );
   112→}
