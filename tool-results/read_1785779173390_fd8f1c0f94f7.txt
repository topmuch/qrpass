   233→// ─── Helper: Parse hotelCoords ───
   234→function parseCoords(coords: { lat: number; lng: number } | string | null): { lat: number; lng: number } | null {
   235→  if (!coords) return null;
   236→  if (typeof coords === 'object' && 'lat' in coords && 'lng' in coords) {
   237→    return { lat: coords.lat, lng: coords.lng };
   238→  }
   239→  if (typeof coords === 'string') {
   240→    try {
   241→      const parsed = JSON.parse(coords);
   242→      if (parsed && typeof parsed === 'object' && 'lat' in parsed && 'lng' in parsed) {
   243→        return { lat: parsed.lat, lng: parsed.lng };
   244→      }
   245→    } catch {
   246→      const parts = coords.split(',').map((s) => parseFloat(s.trim()));
   247→      if (parts.length === 2 && !isNaN(parts[0]) && !isNaN(parts[1])) {
   248→        return { lat: parts[0], lng: parts[1] };
   249→      }
   250→    }
   251→  }
   252→  return null;
   253→}

   256→function cleanPhone(phone: string): string {
   257→  return phone.replace(/[^0-9+]/g, '').replace(/^\+/g, '');
