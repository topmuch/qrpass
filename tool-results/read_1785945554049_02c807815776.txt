    72→// ─── Helper: format date ───
    73→function formatDate(dateStr?: string | null): string {
    74→  if (!dateStr) return '—';
    75→  try {
    76→    const d = new Date(dateStr);
    77→    return d.toLocaleDateString('fr-FR', { day: '2-digit', month: 'long', year: 'numeric' });
    78→  } catch {
    79→    return dateStr;
    80→  }
    81→}
    82→
    83→// ─── Helper: compute remaining time string ───
    84→function computeRemaining(expiryIso: string): string {
    85→  const diff = new Date(expiryIso).getTime() - Date.now();
    86→  if (diff <= 0) return 'Expiré';
    87→  const h = Math.floor(diff / 3600000);
    88→  const m = Math.floor((diff % 3600000) / 60000);
    89→  const s = Math.floor((diff % 60000) / 1000);
    90→  if (h > 0) return `${h}h ${m}m ${s}s`;
    91→  if (m > 0) return `${m}m ${s}s`;
    92→  return `${s}s`;
    93→}
    94→
    95→// ─── Helper: OTP countdown ───
    96→function useOtpCountdown(expiryIso?: string | null) {
    97→  const initialValue = expiryIso ? computeRemaining(expiryIso) : 'Expiré';
    98→  const [remaining, setRemaining] = useState<string>(initialValue);
    99→
   100→  useEffect(() => {
   101→    if (!expiryIso) return;
   102→
   103→    const tick = () => {
   104→      setRemaining(computeRemaining(expiryIso));
   105→    };
   106→
   107→    const interval = setInterval(tick, 1000);
   108→    return () => clearInterval(interval);
   109→  }, [expiryIso]);
   110→
   111→  return remaining;
   112→}
