    53→    53→
    54→    54→export default function EtiquettesPage() {
    55→    55→  const qrRef = useRef<HTMLDivElement>(null);
    56→    56→
    57→    57→  const [sets, setSets] = useState<QRSet[]>([]);
    58→    58→  const [agencyGroups, setAgencyGroups] = useState<AgencyGroup[]>([]);
    59→    59→  const [stats, setStats] = useState<Stats>({
    60→    60→    totalSets: 0,
    61→    61→    totalQr: 0,
    62→    62→    hajjSets: 0,
    63→    63→
    64→    64→  });

    67→    67→  const [activeTab, setActiveTab] = useState<'hajj'>('hajj');
    68→    68→
    69→    69→  // Modals
    70→    70→  const [showDetailModal, setShowDetailModal] = useState(false);
    71→    71→  const [showDeleteModal, setShowDeleteModal] = useState(false);
    72→    72→  const [selectedSet, setSelectedSet] = useState<QRSet | null>(null);
    73→    73→  const [isDownloading, setIsDownloading] = useState(false);
    74→    74→
    75→    75→  useEffect(() => {
    76→    76→    fetchSets();
    77→    77→  }, [activeTab, search]);
    78→    78→
    79→    79→  const fetchSets = async () => {
    80→    80→    setLoading(true);
    81→    81→    try {
    82→    82→      const params = new URLSearchParams();
    83→    83→      params.set('type', activeTab);
    84→    84→      if (search) params.set('search', search);
    85→    85→
    86→    86→      const response = await fetch(`/api/qrcodes?${params}`);
    87→    87→      const data = await response.json();

    89→    89→      // Calculate activation status for each set
    90→    90→      const setsWithStatus = data.sets.map((set: QRSet) => ({
    91→    91→        ...set,
    92→    92→        activationStatus: getActivationStatus(set.status)
    93→    93→      }));
    94→    94→
    95→    95→      // Filter to only show sets matching the active tab type
    96→    96→      const filteredSets = setsWithStatus.filter((set: QRSet) => set.type === activeTab);
    97→    97→      
    98→    98→      setSets(filteredSets);
    99→    99→      setStats(data.stats);
   100→   100→
   101→   101→      // Group by agency
   102→   102→      const groupedByAgency = groupByAgency(filteredSets);
   103→   103→      setAgencyGroups(groupedByAgency);
   104→   104→    } catch (error) {
   105→   105→      console.error('Error fetching QR sets:', error);
   106→   106→    } finally {
   107→   107→      setLoading(false);
   108→   108→    }
   109→   109→  };
   110→   110→
   111→   111→  const groupByAgency = (sets: QRSet[]): AgencyGroup[] => {
   112→   112→    const groups = new Map<string | null, AgencyGroup>();
   113→   113→
   114→   114→    sets.forEach(set => {
   115→   115→      const key = set.agencyId || 'no-agency';
   116→   116→      const agencyName = set.agencyName || 'Sans agence';
   117→   117→
   118→   118→      if (!groups.has(key)) {
   119→   119→        groups.set(key, {
   120→   120→          agencyId: set.agencyId,
   121→   121→          agencyName,
   122→   122→          sets: [],
   123→   123→          totalQr: 0,
   124→   124→          isExpanded: true, // Expanded by default
   125→   125→        });
   126→   126→      }
   127→   127→
   128→   128→      const group = groups.get(key)!;
   129→   129→      group.sets.push(set);
   130→   130→      group.totalQr += set.qrCount;
   131→   131→    });
   132→   132→
   133→   133→    // Sort: agencies with name first, then "Sans agence"
   134→   134→    return Array.from(groups.values()).sort((a, b) => {
   135→   135→      if (a.agencyId === null) return 1;
   136→   136→      if (b.agencyId === null) return -1;
   137→   137→      return a.agencyName.localeCompare(b.agencyName);
   138→   138→    });
   139→   139→  };
   140→   140→
   141→   141→  const getActivationStatus = (status: string): 'new' | 'partial' | 'activated' => {
   142→   142→    if (status === 'active' || status === 'scanned') return 'activated';
   143→   143→    if (status === 'partial') return 'partial';
   144→   144→    return 'new';
   145→   145→  };
   146→   146→
   147→   147→  const toggleAgencyGroup = (agencyId: string | null) => {
   148→   148→    setAgencyGroups(prev => 
   149→   149→      prev.map(group => 
   150→   150→        group.agencyId === agencyId 
   151→   151→          ? { ...group, isExpanded: !group.isExpanded }
   152→   152→          : group
   153→   153→      )
   154→   154→    );
   155→   155→  };
   156→   156→
   157→   157→  const handleDeleteSet = async () => {
   158→   158→    if (!selectedSet) return;
   159→   159→
   160→   160→    try {
   161→   161→      const params = new URLSearchParams({ setId: selectedSet.setId });
   162→   162→      const response = await fetch(`/api/qrcodes?${params}`, { method: 'DELETE' });
   163→   163→      const data = await response.json();
   164→   164→
   165→   165→      if (response.ok && data.success) {
   166→   166→        fetchSets();
   167→   167→        setShowDeleteModal(false);
   168→   168→        setSelectedSet(null);
   169→   169→      } else {
   170→   170→        alert(`Erreur: ${data.error || 'Erreur lors de la suppression'}`);
   171→   171→      }
   172→   172→    } catch (error) {
   173→   173→      console.error('Error deleting set:', error);
   174→   174→      alert('Erreur lors de la suppression');
   175→   175→    }
   176→   176→  };
   177→   177→
   178→   178→  const handleDownloadPDF = async (set: QRSet) => {
   179→   179→    setIsDownloading(true);
   180→   180→    setSelectedSet(set);
   181→   181→
   182→   182→    try {
   183→   183→      const canvas = document.createElement('canvas');
   184→   184→      const ctx = canvas.getContext('2d');
   185→   185→      if (!ctx) throw new Error('Canvas not supported');
   186→   186→
   187→   187→      const qrSize = 300;
   188→   188→      const padding = 40;
   189→   189→      const headerHeight = 100;
   190→   190→
   191→   191→      const cols = Math.min(set.qrCount, 3);
   192→   192→      const rows = Math.ceil(set.qrCount / 3);
   193→   193→
   194→   194→      canvas.width = cols * qrSize + (cols + 1) * padding;
   195→   195→      canvas.height = headerHeight + rows * qrSize + (rows + 1) * padding + 60;
   196→   196→
   197→   197→      // Background
   198→   198→      ctx.fillStyle = '#ffffff';
   199→   199→      ctx.fillRect(0, 0, canvas.width, canvas.height);
   200→   200→
   201→   201→      // Header
   202→   202→      ctx.fillStyle = set.type === 'hajj' ? '#1D4ED8' : '#f59e0b';
   203→   203→      ctx.fillRect(0, 0, canvas.width, headerHeight);
   204→   204→
   205→   205→      ctx.fillStyle = '#ffffff';
   206→   206→      ctx.font = 'bold 24px Arial';
   207→   207→      ctx.textAlign = 'center';
   208→   208→      ctx.fillText('PassHajj - Étiquettes', canvas.width / 2, 35);
   209→   209→
   210→   210→      ctx.font = '16px Arial';
   211→   211→      ctx.fillText(`${set.setId} | Hajj 2026 | ${set.qrCount} QR`, canvas.width / 2, 65);
   212→   212→      if (set.agencyName) {
   213→   213→        ctx.font = '14px Arial';
   214→   214→        ctx.fillText(`Agence: ${set.agencyName}`, canvas.width / 2, 85);
   215→   215→      }
   216→   216→
   217→   217→      // Generate QR codes
   218→   218→      for (let i = 0; i < set.references.length; i++) {
   219→   219→        const col = i % 3;
   220→   220→        const row = Math.floor(i / 3);
   221→   221→        const x = padding + col * (qrSize + padding);
   222→   222→        const y = headerHeight + padding + row * (qrSize + padding);
   223→   223→
   224→   224→        ctx.fillStyle = '#ffffff';
   225→   225→        ctx.fillRect(x, y, qrSize, qrSize);
   226→   226→
   227→   227→        const qrUrl = `${window.location.origin}/found/${set.references[i]}`;
   228→   228→        const svgElement = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
   229→   229→        const qrSvg = <QRCodeSVG value={qrUrl} size={qrSize - 40} level="H" fgColor={set.type === 'hajj' ? '#1D4ED8' : '#f59e0b'} />;
   230→   230→
   231→   231→        // Use a promise-based approach to render SVG to canvas
   232→   232→        const svgString = `<svg xmlns="http://www.w3.org/2000/svg" width="${qrSize - 40}" height="${qrSize - 40}">
   233→   233→          <rect width="100%" height="100%" fill="white"/>
   234→   234→          ${generateQRPath(qrUrl, qrSize - 40, set.type === 'hajj' ? '#1D4ED8' : '#f59e0b')}
   235→   235→        </svg>`;
   236→   236→
   237→   237→        const img = new Image();
   238→   238→        const svgBlob = new Blob([svgString], { type: 'image/svg+xml;charset=utf-8' });
   239→   239→        const url = URL.createObjectURL(svgBlob);
   240→   240→
   241→   241→        await new Promise<void>((resolve) => {
   242→   242→          img.onload = () => {
   243→   243→            ctx.drawImage(img, x + 20, y + 20, qrSize - 40, qrSize - 40);
   244→   244→            URL.revokeObjectURL(url);
   245→   245→            resolve();
   246→   246→          };
   247→   247→          img.onerror = () => {
   248→   248→            URL.revokeObjectURL(url);
   249→   249→            resolve();
   250→   250→          };
   251→   251→          img.src = url;
   252→   252→        });
   253→   253→
   254→   254→        // Reference label
   255→   255→        ctx.fillStyle = '#1e293b';
   256→   256→        ctx.font = 'bold 12px Arial';
   257→   257→        ctx.textAlign = 'center';
   258→   258→        ctx.fillText(set.references[i], x + qrSize / 2, y + qrSize - 15);
   259→   259→      }
   260→   260→
   261→   261→      // Footer
   262→   262→      ctx.fillStyle = '#64748b';
   263→   263→      ctx.font = '12px Arial';
   264→   264→      ctx.textAlign = 'center';
   265→   265→      ctx.fillText(`Généré le ${new Date().toLocaleDateString('fr-FR')} | QRPass.com`, canvas.width / 2, canvas.height - 20);
   266→   266→
   267→   267→      // Download
   268→   268→      const link = document.createElement('a');
   269→   269→      link.download = `PassHajj-${set.setId}.png`;
   270→   270→      link.href = canvas.toDataURL('image/png');
   271→   271→      link.click();
   272→   272→
   273→   273→      setIsDownloading(false);
   274→   274→    } catch (error) {
   275→   275→      console.error('Error generating PDF:', error);
   276→   276→      setIsDownloading(false);
   277→   277→      alert('Erreur lors de la génération');
   278→   278→    }
   279→   279→  };
   280→   280→
   281→   281→  // Simple QR path generator fallback
   282→   282→  const generateQRPath = (url: string, size: number, color: string): string => {
   283→   283→    return `<text x="50%" y="50%" text-anchor="middle" fill="${color}" font-size="10">${url}</text>`;
   284→   284→  };

   286→   286→  const handleShareSet = async (set: QRSet) => {
   287→   287→    const shareUrl = `${window.location.origin}/found/${set.references[0]}`;
   288→   288→    if (navigator.share) {
   289→   289→      try {
   290→   290→        await navigator.share({
   291→   291→          title: `PassHajj - ${set.setId}`,
   292→   292→          text: `${set.qrCount} QR codes pour ${set.agencyName || 'agence'}`,
   293→   293→          url: shareUrl,
   294→   294→        });
   295→   295→      } catch (err) {
   296→   296→        console.log('Share cancelled');
   297→   297→      }
   298→   298→    } else {
   299→   299→      navigator.clipboard.writeText(shareUrl);
   300→   300→      alert('Lien copié !');
   301→   301→    }
   302→   302→  };
   303→   303→
   304→   304→  const formatDate = (date: Date | string) => {
   305→   305→    return new Date(date).toLocaleDateString('fr-FR', {
   306→   306→      day: '2-digit',
   307→   307→      month: 'short',
   308→   308→      year: 'numeric',
   309→   309→    });
   310→   310→  };
   311→   311→
   312→   312→  return (
   313→   313→    <div className="max-w-7xl mx-auto">
   314→   314→      {/* Header */}
   315→   315→      <div className="flex flex-col md:flex-row md:items-center justify-between gap-4 mb-6">
   316→   316→        <div>
   317→   317→          <h1 className="text-2xl font-bold text-slate-800 dark:text-white">Étiquettes QR</h1>
   318→   318→          <p className="text-slate-500 dark:text-slate-400 text-sm">
   319→   319→            {stats.totalSets} sets • {stats.totalQr} QR codes
   320→   320→          </p>
   321→   321→        </div>
   322→   322→
   323→   323→        <div className="flex items-center gap-3">
   324→   324→          <div className="relative">
   325→   325→            <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" />
   326→   326→            <input
   327→   327→              type="text"
   328→   328→              placeholder="Rechercher..."
   329→   329→              value={search}
   330→   330→              onChange={(e) => setSearch(e.target.value)}
   331→   331→              className="pl-10 pr-4 py-2 w-64 bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-xl text-sm focus:outline-none focus:ring-2 focus:ring-green-500"
   332→   332→            />
   333→   333→          </div>
   334→   334→
   335→   335→          <button
   336→   336→            onClick={() => fetchSets()}
   337→   337→            className="p-2 bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-xl hover:bg-slate-50 dark:hover:bg-slate-700 transition-colors"
   338→   338→          >
   339→   339→            <RefreshCw className={`w-4 h-4 text-slate-500 ${loading ? 'animate-spin' : ''}`} />
   340→   340→          </button>
   341→   341→
   342→   342→          <Link
   343→   343→            href="/admin/generer"
   344→   344→            className="flex items-center gap-2 px-4 py-2 bg-green-500 hover:bg-green-600 text-white rounded-xl font-medium transition-colors"
   345→   345→          >
   346→   346→            <QrCode className="w-4 h-4" />
   347→   347→            Générer QR
   348→   348→          </Link>
   349→   349→        </div>
   350→   350→      </div>
   351→   351→
   352→   352→      {/* Tabs */}
   353→   353→      <div className="flex gap-2 mb-6">
   354→   354→        <button
   355→   355→          onClick={() => setActiveTab('hajj')}
   356→   356→          className={`flex items-center gap-2 px-6 py-3 rounded-xl font-medium transition-all ${
   357→   357→            activeTab === 'hajj'
   358→   358→              ? 'bg-blue-700 text-white shadow-lg shadow-blue-700/30'
   359→   359→              : 'bg-white dark:bg-slate-800 text-slate-600 dark:text-slate-300 border border-slate-200 dark:border-slate-700 hover:bg-slate-50 dark:hover:bg-slate-700'
   360→   360→          }`}
   361→   361→        >
   362→   362→          <Luggage className="w-5 h-5" />
   363→   363→          Hajj
   364→   364→          <span className={`px-2 py-0.5 rounded-full text-xs ${
   365→   365→            activeTab === 'hajj' 
   366→   366→              ? 'bg-white/20 text-white' 
   367→   367→              : 'bg-slate-100 dark:bg-slate-700 text-slate-600 dark:text-slate-400'
   368→   368→          }`}>
   369→   369→            {stats.hajjSets}
   370→   370→          </span>
   371→   371→        </button>
   372→   372→      </div>
   373→   373→
   374→   374→      {/* Content */}
   375→   375→      {loading ? (
   376→   376→        <div className="flex items-center justify-center py-20">
   377→   377→          <div className="animate-spin w-8 h-8 border-4 border-slate-200 border-t-green-500 rounded-full" />
   378→   378→        </div>
   379→   379→      ) : agencyGroups.length === 0 ? (
   380→   380→        <div className="bg-white dark:bg-slate-800 rounded-2xl border border-slate-200 dark:border-slate-700 p-12 text-center">
   381→   381→          <div className={`w-16 h-16 mx-auto mb-4 rounded-2xl flex items-center justify-center ${
   382→   382→            activeTab === 'hajj' ? 'bg-emerald-100 dark:bg-blue-600/20' : 'bg-amber-100 dark:bg-blue-600/20'
   383→   383→          }`}>
   384→   384→            {activeTab === 'hajj' ? (
   385→   385→              <Luggage className="w-8 h-8 text-blue-700" />
   386→   386→            ) : (
   387→   387→              <Plane className="w-8 h-8 text-amber-600" />
   388→   388→            )}
   389→   389→          </div>
   390→   390→          <h3 className="text-lg font-semibold text-slate-800 dark:text-white mb-2">
   391→   391→            Aucun QR code Hajj
   392→   392→          </h3>
   393→   393→          <p className="text-slate-500 dark:text-slate-400 mb-4">
   394→   394→            {search ? 'Aucun résultat pour votre recherche' : 'Commencez par générer des QR codes'}
   395→   395→          </p>
   396→   396→          {!search && (
   397→   397→            <Link

   413→   413→              {/* Agency Header */}
   414→   414→              <button
