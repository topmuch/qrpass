    31→// Photos uploaded via /api/pilgrims/upload-photo are stored as /uploads/pilgrim-photos/xxx.jpg
    32→// but the Next.js dev server doesn't serve dynamically created files from public/.
    33→// The /api/serve-upload API serves them, so we rewrite the path.
    34→function getPhotoDisplayUrl(photoUrl: string | null): string | null {
    35→  if (!photoUrl) return null;
    36→  // If it's a relative path under /uploads/, route through the serve-upload API
    37→  if (photoUrl.startsWith('/uploads/')) {
    38→    return `/api/serve-upload/${photoUrl.slice('/uploads/'.length)}`;
    39→  }
    40→  return photoUrl;
    41→}

   299→// ─── Helper: Parse hotelCoords ───
   300→function parseCoords(coords: { lat: number; lng: number } | string | null): { lat: number; lng: number } | null {
   301→  if (!coords) return null;
   302→  if (typeof coords === 'object' && 'lat' in coords && 'lng' in coords) {
   303→    return { lat: coords.lat, lng: coords.lng };
   304→  }
   305→  if (typeof coords === 'string') {
   306→    try {
   307→      const parsed = JSON.parse(coords);
   308→      if (parsed && typeof parsed === 'object' && 'lat' in parsed && 'lng' in parsed) {
   309→        return { lat: parsed.lat, lng: parsed.lng };
   310→      }
   311→    } catch {
   312→      const parts = coords.split(',').map((s) => parseFloat(s.trim()));
   313→      if (parts.length === 2 && !isNaN(parts[0]) && !isNaN(parts[1])) {
   314→        return { lat: parts[0], lng: parts[1] };
   315→      }
   316→    }
   317→  }
   318→  return null;
   319→}
