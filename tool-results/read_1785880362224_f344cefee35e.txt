    29→function getPhotoDisplayUrl(photoUrl: string | null): string | null {
    30→  if (!photoUrl) return null;
    31→  // If it's a relative path under /uploads/, route through the serve-upload API
    32→  if (photoUrl.startsWith('/uploads/')) {
    33→    return `/api/serve-upload/${photoUrl.slice('/uploads/'.length)}`;
    34→  }
    35→  return photoUrl;
    36→}

   294→// ─── Helper: Parse hotelCoords ───
   295→function parseCoords(coords: { lat: number; lng: number } | string | null): { lat: number; lng: number } | null {
   296→  if (!coords) return null;
   297→  if (typeof coords === 'object' && 'lat' in coords && 'lng' in coords) {
   298→    return { lat: coords.lat, lng: coords.lng };
   299→  }
   300→  if (typeof coords === 'string') {
   301→    try {
   302→      const parsed = JSON.parse(coords);
   303→      if (parsed && typeof parsed === 'object' && 'lat' in parsed && 'lng' in parsed) {
   304→        return { lat: parsed.lat, lng: parsed.lng };
   305→      }
   306→    } catch {
   307→      const parts = coords.split(',').map((s) => parseFloat(s.trim()));
   308→      if (parts.length === 2 && !isNaN(parts[0]) && !isNaN(parts[1])) {
   309→        return { lat: parts[0], lng: parts[1] };
   310→      }
   311→    }
   312→  }
   313→  return null;
   314→}
