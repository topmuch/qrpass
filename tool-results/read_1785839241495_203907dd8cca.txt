   286→   286→// ─── Helper: Parse hotelCoords ───
   287→   287→function parseCoords(coords: { lat: number; lng: number } | string | null): { lat: number; lng: number } | null {
   288→   288→  if (!coords) return null;
   289→   289→  if (typeof coords === 'object' && 'lat' in coords && 'lng' in coords) {
   290→   290→    return { lat: coords.lat, lng: coords.lng };
   291→   291→  }
   292→   292→  if (typeof coords === 'string') {
   293→   293→    try {
   294→   294→      const parsed = JSON.parse(coords);
   295→   295→      if (parsed && typeof parsed === 'object' && 'lat' in parsed && 'lng' in parsed) {
   296→   296→        return { lat: parsed.lat, lng: parsed.lng };
   297→   297→      }
   298→   298→    } catch {
   299→   299→      const parts = coords.split(',').map((s) => parseFloat(s.trim()));
   300→   300→      if (parts.length === 2 && !isNaN(parts[0]) && !isNaN(parts[1])) {
   301→   301→        return { lat: parts[0], lng: parts[1] };
   302→   302→      }
   303→   303→    }
   304→   304→  }
   305→   305→  return null;
   306→   306→}
