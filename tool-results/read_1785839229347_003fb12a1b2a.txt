   286→// ─── Helper: Parse hotelCoords ───
   287→function parseCoords(coords: { lat: number; lng: number } | string | null): { lat: number; lng: number } | null {
   288→  if (!coords) return null;
   289→  if (typeof coords === 'object' && 'lat' in coords && 'lng' in coords) {
   290→    return { lat: coords.lat, lng: coords.lng };
   291→  }
   292→  if (typeof coords === 'string') {
   293→    try {
   294→      const parsed = JSON.parse(coords);
   295→      if (parsed && typeof parsed === 'object' && 'lat' in parsed && 'lng' in parsed) {
   296→        return { lat: parsed.lat, lng: parsed.lng };
   297→      }
   298→    } catch {
   299→      const parts = coords.split(',').map((s) => parseFloat(s.trim()));
   300→      if (parts.length === 2 && !isNaN(parts[0]) && !isNaN(parts[1])) {
   301→        return { lat: parts[0], lng: parts[1] };
   302→      }
   303→    }
   304→  }
   305→  return null;
   306→}
