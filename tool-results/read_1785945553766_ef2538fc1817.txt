    79→    79→    case 'completed':
    80→    80→      return (
    81→    81→        <Badge className="bg-blue-100 text-blue-700 border-blue-200 hover:bg-blue-100">
    82→    82→          Terminé

    88→    88→          Annulé

   119→   119→function formatDate(dateStr: string | null | undefined): string {
   120→   120→  if (!dateStr) return '—';
   121→   121→  try {
   122→   122→    const d = new Date(dateStr);
   123→   123→    return d.toLocaleDateString('fr-FR', {
   124→   124→      day: '2-digit',
   125→   125→      month: 'short',
   126→   126→      year: 'numeric',
   127→   127→    });
   128→   128→  } catch {
   129→   129→    return '—';
   130→   130→  }
