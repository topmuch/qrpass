     1→'use client';
     2→
     3→import { useState, useEffect, useRef, useCallback } from 'react';
     4→import Link from 'next/link';
     5→import { QRCodeSVG } from 'qrcode.react';
     6→import {
     7→  Search,
     8→  Eye,
     9→  Download,
    10→  Share2,
    11→  Trash2,
    12→  Plane,
    13→  Luggage,
    14→  QrCode,
    15→  Plus,
    16→  ArrowLeft,
    17→  X,
    18→  AlertTriangle,
    19→  CheckCircle,
    20→  Clock,
    21→  FileArchive,
    22→  Archive,
    23→  Filter,
    24→  Loader2,
    25→} from "lucide-react";
    26→
    27→interface QRSet {
    28→  id: string;
    29→  setId: string;
    30→  type: string;
    31→  agencyId: string | null;
    32→  agencyName: string | null;
    33→  createdAt: Date;
    34→  qrCount: number;
    35→  references: string[];
    36→  status: string;
    37→  travelerName: string | null;
    38→  baggageIds: string[];
    39→}
    40→
    41→interface Stats {
    42→  totalSets: number;
    43→  totalQr: number;
    44→  hajjSets: number;
    45→}
    46→
    47→interface Agency {
    48→  id: string;
    49→  name: string;
    50→}
    51→
    52→export default function QRCodesPage() {
    53→  const qrRef = useRef<HTMLDivElement>(null);
    54→  
    55→  const [sets, setSets] = useState<QRSet[]>([]);
    56→  const [stats, setStats] = useState<Stats>({
    57→    totalSets: 0,
    58→    totalQr: 0,
    59→    hajjSets: 0,
    60→
    61→  });
    62→  const [loading, setLoading] = useState(true);
    63→  const [search, setSearch] = useState('');
    64→  const [typeFilter, setTypeFilter] = useState('all');
    65→  
    66→  // Modals
    67→  const [showDetailModal, setShowDetailModal] = useState(false);
    68→  const [showDeleteModal, setShowDeleteModal] = useState(false);
    69→  const [showExportModal, setShowExportModal] = useState(false);
    70→  const [selectedSet, setSelectedSet] = useState<QRSet | null>(null);
    71→  const [isDownloading, setIsDownloading] = useState(false);
    72→
    73→  // Selection for bulk export
    74→  const [selectedSetIds, setSelectedSetIds] = useState<Set<string>>(new Set());
    75→  const [selectAll, setSelectAll] = useState(false);
    76→
    77→  // Export modal state
    78→  const [exportAgencies, setExportAgencies] = useState<Agency[]>([]);
    79→  const [exportForm, setExportForm] = useState({
    80→    mode: 'selected' as 'selected' | 'agency' | 'type',
    81→    agencyId: '',
    82→    type: 'hajj' as 'hajj',
    83→  });
    84→  const [isExporting, setIsExporting] = useState(false);
    85→  const [exportProgress, setExportProgress] = useState('');
    86→
    87→  useEffect(() => {
    88→    fetchSets();
    89→    fetchAgencies();
    90→  }, [typeFilter, search]);
    91→
    92→  const fetchSets = async () => {
    93→    try {
    94→      const params = new URLSearchParams();
    95→      if (typeFilter !== 'all') params.set('type', typeFilter);
    96→      if (search) params.set('search', search);
    97→
    98→      const response = await fetch(`/api/qrcodes?${params}`);
    99→      const data = await response.json();
   100→
   101→      setSets(data.sets);
   102→      setStats(data.stats);
   103→    } catch (error) {
   104→      console.error('Error fetching QR sets:', error);
   105→    } finally {
   106→      setLoading(false);
   107→    }
   108→  };
   109→
   110→  const fetchAgencies = async () => {
   111→    try {
   112→      const res = await fetch('/api/admin/agencies');
   113→      const data = await res.json();
   114→      setExportAgencies(data.agencies || []);
   115→    } catch (error) {
   116→      console.error('Error fetching agencies:', error);
   117→    }
   118→  };
   119→
   120→  // Selection handlers
   121→  const toggleSetSelection = useCallback((setId: string) => {
   122→    setSelectedSetIds(prev => {
   123→      const next = new Set(prev);
   124→      if (next.has(setId)) {
   125→        next.delete(setId);
   126→      } else {
   127→        next.add(setId);
   128→      }
   129→      return next;
   130→    });
   131→  }, []);
   132→
   133→  const toggleSelectAll = useCallback(() => {
   134→    if (selectAll) {
   135→      setSelectedSetIds(new Set());
   136→      setSelectAll(false);
   137→    } else {
   138→      setSelectedSetIds(new Set(sets.map(s => s.setId)));
   139→      setSelectAll(true);
   140→    }
   141→  }, [selectAll, sets]);
   142→
   143→  // Keep selectAll in sync
   144→  useEffect(() => {
   145→    if (sets.length > 0 && selectedSetIds.size === sets.length) {
   146→      setSelectAll(true);
   147→    } else {
   148→      setSelectAll(false);
   149→    }
   150→  }, [selectedSetIds, sets.length]);
   151→
   152→  const handleDeleteSet = async () => {
   153→    if (!selectedSet) return;
   154→
   155→    try {
   156→      const params = new URLSearchParams({ setId: selectedSet.setId });
   157→      await fetch(`/api/qrcodes?${params}`, { method: 'DELETE' });
   158→
   159→      setSets(sets.filter(s => s.setId !== selectedSet.setId));
   160→      setShowDeleteModal(false);
   161→      setSelectedSet(null);
   162→    } catch (error) {
   163→      console.error('Error deleting set:', error);
   164→    }
   165→  };
   166→
   167→  const handleDownloadSet = async (set: QRSet) => {
   168→    setIsDownloading(true);
   169→    setSelectedSet(set);
   170→
   171→    try {
   172→      // Use server-side ZIP export for single set with timeout
   173→      const controller = new AbortController();
   174→      const timeoutId = setTimeout(() => controller.abort(), 120000); // 2 min timeout
   175→
   176→      let response: Response;
   177→      try {
   178→        response = await fetch('/api/admin/baggages/export-zip', {
   179→          method: 'POST',
   180→          headers: { 'Content-Type': 'application/json' },
   181→          body: JSON.stringify({ setId: set.setId }),
   182→          signal: controller.signal,
   183→        });
   184→      } catch (fetchError) {
   185→        clearTimeout(timeoutId);
   186→        if (fetchError instanceof DOMException && fetchError.name === 'AbortError') {
   187→          throw new Error('Délai d\'attente dépassé');
   188→        }
   189→        throw fetchError;
   190→      }
   191→      clearTimeout(timeoutId);
   192→
   193→      if (!response.ok) {
   194→        const errorData = await response.json().catch(() => ({ error: 'Export échoué' }));
   195→        throw new Error(errorData.error || 'Export failed');
   196→      }
   197→
   198→      // Get the filename from Content-Disposition header
   199→      const contentDisposition = response.headers.get('Content-Disposition');
   200→      let filename = `QRPass-${set.setId}.zip`;
   201→      if (contentDisposition) {
   202→        const match = contentDisposition.match(/filename\*?=(?:UTF-8'')?([^;]+)/i) ||
   203→                      contentDisposition.match(/filename="?([^"]+)"?/);
   204→        if (match) filename = decodeURIComponent(match[1].replace(/"/g, ''));
   205→      }
   206→
   207→      // Download the ZIP
   208→      const blob = await response.blob();
   209→
   210→      if (blob.size === 0) {
   211→        throw new Error('Le fichier ZIP est vide');
   212→      }
   213→
   214→      const url = URL.createObjectURL(blob);
   215→      const link = document.createElement('a');
   216→      link.href = url;
   217→      link.download = filename;
   218→      document.body.appendChild(link);
   219→      link.click();
   220→      document.body.removeChild(link);
   221→      setTimeout(() => URL.revokeObjectURL(url), 5000);
   222→    } catch (error) {
   223→      console.error('Error downloading:', error);
   224→      alert('Erreur lors du téléchargement: ' + (error instanceof Error ? error.message : 'Erreur inconnue'));
   225→    } finally {
   226→      setIsDownloading(false);
   227→      setSelectedSet(null);
   228→    }
   229→  };
   230→
   231→  // Bulk ZIP export
   232→  const handleExportZip = async () => {
   233→    setIsExporting(true);
   234→    setExportProgress('Préparation de l\'export...');
   235→
   236→    try {
   237→      let payload: Record<string, unknown>;
   238→
   239→      if (exportForm.mode === 'selected') {
   240→        if (selectedSetIds.size === 0) {
   241→          alert('Veuillez sélectionner au moins un set de QR codes');
   242→          setIsExporting(false);
   243→          return;
   244→        }
   245→        payload = { setIds: Array.from(selectedSetIds) };
   246→      } else if (exportForm.mode === 'agency') {
   247→        if (!exportForm.agencyId) {
   248→          alert('Veuillez sélectionner une agence');
   249→          setIsExporting(false);
   250→          return;
   251→        }
   252→        payload = { agencyId: exportForm.agencyId };
   253→      } else {
   254→        payload = { type: exportForm.type, agencyId: '__all__' };
   255→      }
   256→
   257→      setExportProgress('Génération des QR codes...');
   258→
   259→      // Use timeout for large exports
   260→      const controller = new AbortController();
   261→      const timeoutId = setTimeout(() => controller.abort(), 300000); // 5 min timeout
   262→
   263→      let response: Response;
   264→      try {
   265→        response = await fetch('/api/admin/baggages/export-zip', {
   266→          method: 'POST',
   267→          headers: { 'Content-Type': 'application/json' },
   268→          body: JSON.stringify(payload),
   269→          signal: controller.signal,
   270→        });
   271→      } catch (fetchError) {
   272→        clearTimeout(timeoutId);
   273→        if (fetchError instanceof DOMException && fetchError.name === 'AbortError') {
   274→          throw new Error('Délai d\'attente dépassé. L\'export est trop volumineux.');
   275→        }
   276→        throw fetchError;
   277→      }
   278→      clearTimeout(timeoutId);
   279→
   280→      if (!response.ok) {
   281→        const errorData = await response.json().catch(() => ({ error: 'Export échoué' }));
   282→        throw new Error(errorData.error || 'Export failed');
   283→      }
   284→
   285→      setExportProgress('Téléchargement du fichier ZIP...');
   286→
   287→      // Get filename
   288→      const contentDisposition = response.headers.get('Content-Disposition');
   289→      let filename = 'QRPass-export.zip';
   290→      if (contentDisposition) {
   291→        const match = contentDisposition.match(/filename\*?=(?:UTF-8'')?([^;]+)/i) ||
   292→                      contentDisposition.match(/filename="?([^"]+)"?/);
   293→        if (match) filename = decodeURIComponent(match[1].replace(/"/g, ''));
   294→      }
   295→
   296→      const blob = await response.blob();
   297→
   298→      if (blob.size === 0) {
   299→        throw new Error('Le fichier ZIP est vide');
   300→      }
   301→
   302→      const url = URL.createObjectURL(blob);
   303→      const link = document.createElement('a');
   304→      link.href = url;
   305→      link.download = filename;
   306→      document.body.appendChild(link);
   307→      link.click();
   308→      document.body.removeChild(link);
   309→      setTimeout(() => URL.revokeObjectURL(url), 5000);
   310→
   311→      setExportProgress('Export terminé !');
   312→      setTimeout(() => {
   313→        setShowExportModal(false);
   314→        setExportProgress('');
   315→        setSelectedSetIds(new Set());
   316→      }, 1500);
   317→    } catch (error) {
   318→      console.error('Error exporting ZIP:', error);
   319→      alert('Erreur lors de l\'export ZIP: ' + (error instanceof Error ? error.message : 'Erreur inconnue'));
   320→    } finally {
   321→      setIsExporting(false);
   322→    }
   323→  };
   324→
   325→  const handleShareSet = async (set: QRSet) => {
   326→    const shareText = `QRPass - ${set.setId}\n${set.qrCount} QR codes générés\nType: Hajj 2026`;
   327→
   328→    if (navigator.share) {
   329→      try {
   330→        await navigator.share({
   331→          title: `QRPass - ${set.setId}`,
   332→          text: shareText,
   333→          url: window.location.href,
   334→        });
   335→      } catch (err) {
   336→        console.log('Share cancelled');
   337→      }
   338→    } else {
   339→      navigator.clipboard.writeText(shareText);
   340→      alert('Copié dans le presse-papiers !');
   341→    }
   342→  };
   343→
   344→  const formatDate = (date: Date | string) => {
   345→    return new Date(date).toLocaleDateString('fr-FR', {
   346→      day: '2-digit',
   347→      month: '2-digit',
   348→      year: 'numeric',
   349→    });
   350→  };
   351→
   352→  // Filter buttons
   353→  const filterButtons = [
   354→    { id: 'all', label: 'Tous' },
   355→    { id: 'hajj', label: 'Hajj' },
   356→  ];
   357→
   358→  // KPI Cards
   359→  const kpiCards = [
   360→    { title: 'Total Sets', value: stats.totalSets, icon: QrCode, color: 'text-[#b8860b]' },
   361→    { title: 'Total QR', value: stats.totalQr, icon: Luggage, color: 'text-white' },
   362→    { title: 'Hajj', value: stats.hajjSets, icon: Plane, color: 'text-green-400' },
   363→  ];
   364→
   365→  // Calculate total QR in selection
   366→  const selectedQrCount = sets
   367→    .filter(s => selectedSetIds.has(s.setId))
   368→    .reduce((sum, s) => sum + s.qrCount, 0);
   369→
   370→  return (
   371→    <div className="min-h-screen bg-[#080c1a]">
   372→      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-6">
   373→        {/* Header */}
   374→        <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center mb-8 gap-4">
   375→          <div className="flex items-center gap-4">
   376→            <Link
   377→              href="/admin"
   378→              className="p-2 rounded-lg bg-[#0d152a] hover:bg-[#1a2238] transition-colors"
   379→            >
   380→              <ArrowLeft className="w-5 h-5 text-[#a0a8b8]" />
   381→            </Link>
   382→            <div>
   383→              <h1 className="text-2xl font-bold text-white">QR Codes Générés</h1>
   384→              <p className="text-[#a0a8b8] text-sm">Gérez vos sets de QR codes</p>
   385→            </div>
   386→          </div>
   387→          <div className="flex gap-3">
   388→            {/* Export ZIP Button */}
   389→            <button
   390→              onClick={() => setShowExportModal(true)}
   391→              className="inline-flex items-center gap-2 px-4 py-2 bg-gradient-to-r from-[#1e7e34] to-[#0d5e34] text-white rounded-lg hover:from-[#228b22] hover:to-[#1e7e34] transition-all shadow-lg shadow-green-900/30"
   392→            >
   393→              <Archive className="w-4 h-4" />
   394→              Export ZIP
   395→            </button>
   396→            <Link
   397→              href="/admin/generer"
   398→              className="inline-flex items-center gap-2 px-4 py-2 bg-[#b8860b] text-white rounded-lg hover:bg-[#3b82f6] transition-colors"
   399→            >
   400→              <Plus className="w-4 h-4" />
   401→              Générer nouveau
   402→            </Link>
   403→          </div>
   404→        </div>
   405→
   406→        {/* KPI Cards */}
   407→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mb-8">
   408→          {kpiCards.map((card, index) => (
   409→            <div
   410→              key={index}
   411→              className="bg-[#0d152a] border border-[#1a2238] rounded-xl p-5 shadow-sm"
   412→            >
   413→              <div className="flex items-center justify-between mb-2">
   414→                <card.icon className={`w-5 h-5 ${card.color}`} />
   415→              </div>
   416→              <p className="text-2xl font-bold text-white">{card.value}</p>
   417→              <p className="text-[#a0a8b8] text-sm">{card.title}</p>
   418→            </div>
   419→          ))}
   420→        </div>
   421→
   422→        {/* Selection Info Bar */}
   423→        {selectedSetIds.size > 0 && (
   424→          <div className="mb-4 px-5 py-3 bg-gradient-to-r from-[#0d5e34] to-[#1e7e34] rounded-xl border border-[#0d5e34] flex items-center justify-between">
   425→            <div className="flex items-center gap-3">
   426→              <CheckCircle className="w-5 h-5 text-white" />
   427→              <span className="text-white font-medium">
   428→                {selectedSetIds.size} set(s) sélectionné(s) — {selectedQrCount} QR codes
   429→              </span>
   430→            </div>
   431→            <div className="flex gap-2">
   432→              <button
   433→                onClick={() => {
   434→                  setShowExportModal(true);
   435→                  setExportForm(prev => ({ ...prev, mode: 'selected' }));
   436→                }}
   437→                className="px-3 py-1.5 bg-white text-[#0d5e34] rounded-lg text-sm font-medium hover:bg-gray-100 transition-colors flex items-center gap-1"
   438→              >
   439→                <Archive className="w-3.5 h-3.5" />
   440→                Exporter ZIP
   441→              </button>
   442→              <button
   443→                onClick={() => setSelectedSetIds(new Set())}
   444→                className="px-3 py-1.5 bg-white/10 text-white rounded-lg text-sm hover:bg-white/20 transition-colors"
   445→              >
   446→                Tout désélectionner
   447→              </button>
   448→            </div>
   449→          </div>
   450→        )}
   451→
   452→        {/* Search & Filters */}
   453→        <div className="flex flex-col sm:flex-row gap-4 mb-6">
   454→          <div className="relative flex-1">
   455→            <input
   456→              type="text"
   457→              placeholder="Rechercher par référence ou set..."
   458→              value={search}
   459→              onChange={(e) => setSearch(e.target.value)}
   460→              className="w-full bg-[#0d152a] border border-[#1a2238] rounded-lg py-3 px-4 text-[#e0e6f0] placeholder-[#a0a8b8] focus:outline-none focus:border-[#b8860b]/50"
   461→            />
   462→            <Search className="absolute right-4 top-1/2 -translate-y-1/2 w-4 h-4 text-[#a0a8b8]" />
   463→          </div>
   464→          <div className="flex gap-2">
   465→            {filterButtons.map((btn) => (
   466→              <button
   467→                key={btn.id}
   468→                onClick={() => setTypeFilter(btn.id)}
   469→                className={`px-4 py-2 rounded-lg text-sm font-medium transition-colors ${
   470→                  typeFilter === btn.id
   471→                    ? 'bg-[#b8860b] text-white'
   472→                    : 'bg-[#0d152a] text-[#a0a8b8] hover:bg-[#1a2238]'
   473→                }`}
   474→              >
   475→                {btn.label}
   476→              </button>
   477→            ))}
   478→          </div>
   479→        </div>
   480→
   481→        {/* QR Sets List */}
   482→        <div className="space-y-3">
   483→          {loading ? (
   484→            <div className="text-center py-12">
   485→              <div className="w-12 h-12 border-2 border-[#b8860b]/30 border-t-[#b8860b] rounded-full animate-spin mx-auto mb-4" />
   486→              <p className="text-[#a0a8b8]">Chargement...</p>
   487→            </div>
   488→          ) : sets.length === 0 ? (
   489→            <div className="text-center py-12">
   490→              <div className="w-16 h-16 bg-[#0d152a] rounded-full flex items-center justify-center mx-auto mb-4">
   491→                <QrCode className="w-8 h-8 text-[#a0a8b8]" />
   492→              </div>
   493→              <p className="text-[#a0a8b8]">Aucun QR code généré</p>
   494→              <p className="text-sm text-[#a0a8b8]/60 mt-2">
   495→                Générez vos premiers QR codes pour commencer.
   496→              </p>
   497→            </div>
   498→          ) : (
   499→            <>
   500→              {/* Select All Row */}
   501→              <div className="flex items-center gap-3 px-5 py-2 bg-[#0d152a] rounded-lg border border-[#1a2238]">
   502→                <input
   503→                  type="checkbox"
   504→                  checked={selectAll}
   505→                  onChange={toggleSelectAll}
   506→                  className="w-4 h-4 rounded border-[#1a2238] bg-[#080c1a] text-[#b8860b] focus:ring-[#b8860b] accent-[#b8860b]"
   507→                />
   508→                <span className="text-[#a0a8b8] text-sm">
   509→                  Tout sélectionner ({sets.length} sets)
   510→                </span>
   511→              </div>
   512→
   513→              {sets.map((set) => (
   514→                <div
   515→                  key={set.id}
   516→                  className={`flex flex-col sm:flex-row items-start sm:items-center justify-between p-5 rounded-xl border gap-4 transition-colors ${
   517→                    selectedSetIds.has(set.setId)
   518→                      ? 'bg-[#0d5e34]/10 border-[#0d5e34]/50'
   519→                      : 'bg-[#0a0f2c] border-[#1a1a3a]'
   520→                  }`}
   521→                >
   522→                  {/* Left: Checkbox + Info */}
   523→                  <div className="flex items-start gap-4">
   524→                    <input
   525→                      type="checkbox"
   526→                      checked={selectedSetIds.has(set.setId)}
   527→                      onChange={() => toggleSetSelection(set.setId)}
   528→                      className="mt-1 w-4 h-4 rounded border-[#1a2238] bg-[#080c1a] text-[#b8860b] focus:ring-[#b8860b] accent-[#b8860b]"
   529→                    />
   530→                    <div className={`w-10 h-10 rounded-lg flex items-center justify-center ${
   531→                      set.type === 'identity' ? 'bg-emerald-600' : 'bg-[#1e3a8a]'
   532→                    }`}>
   533→                      {set.type === 'identity' ? (
   534→                        <span className="text-lg">👤</span>
   535→                      ) : (
   536→                        <span className="text-lg">🧳</span>
   537→                      )}
   538→                    </div>
   539→
   540→                    <div>
   541→                      <div className="flex items-center gap-2 mb-1">
   542→                        <h3 className="font-bold text-white">{set.setId}</h3>
   543→                        <span className={`px-2 py-0.5 text-[#e0e6f0] text-xs rounded ${
   544→                          set.type === 'identity' ? 'bg-emerald-600' : 'bg-[#1e3a8a]'
   545→                        }`}>
   546→                          {set.type === 'identity' ? '👤 Identity' : '🧳 Bagage'}
   547→                        </span>
   548→                      </div>
   549→                      <div className="text-[#a0a8b8] text-sm flex flex-wrap gap-3">
   550→                        <span>👤 {set.travelerName || '1 pèlerin'}</span>
