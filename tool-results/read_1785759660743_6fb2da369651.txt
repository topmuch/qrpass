    52→    52→
    53→    53→export default function EtiquettesPage() {
    54→    54→  const qrRef = useRef<HTMLDivElement>(null);
    55→    55→
    56→    56→  const [sets, setSets] = useState<QRSet[]>([]);
    57→    57→  const [agencyGroups, setAgencyGroups] = useState<AgencyGroup[]>([]);
    58→    58→  const [stats, setStats] = useState<Stats>({
    59→    59→    totalSets: 0,
    60→    60→    totalQr: 0,
    61→    61→    hajjSets: 0,
    62→    62→
    63→    63→  });
    64→    64→  const [loading, setLoading] = useState(true);
    65→    65→  const [search, setSearch] = useState('');
    66→    66→  const [activeTab, setActiveTab] = useState<'hajj'>('hajj');
    67→    67→
    68→    68→  // Modals
    69→    69→  const [showDetailModal, setShowDetailModal] = useState(false);
    70→    70→  const [showDeleteModal, setShowDeleteModal] = useState(false);
    71→    71→  const [selectedSet, setSelectedSet] = useState<QRSet | null>(null);
    72→    72→  const [isDownloading, setIsDownloading] = useState(false);
    73→    73→
    74→    74→  useEffect(() => {
    75→    75→    fetchSets();
    76→    76→  }, [activeTab, search]);
    77→    77→
    78→    78→  const fetchSets = async () => {
    79→    79→    setLoading(true);
    80→    80→    try {
    81→    81→      const params = new URLSearchParams();
    82→    82→      params.set('type', activeTab);
    83→    83→      if (search) params.set('search', search);
    84→    84→
    85→    85→      const response = await fetch(`/api/qrcodes?${params}`);
    86→    86→      const data = await response.json();
    87→    87→
    88→    88→      // Calculate activation status for each set
    89→    89→      const setsWithStatus = data.sets.map((set: QRSet) => ({
    90→    90→        ...set,
    91→    91→        activationStatus: getActivationStatus(set.status)
    92→    92→      }));
    93→    93→
    94→    94→      // Filter to only show sets matching the active tab type
    95→    95→      const filteredSets = setsWithStatus.filter((set: QRSet) => set.type === activeTab);
    96→    96→      
    97→    97→      setSets(filteredSets);
    98→    98→      setStats(data.stats);
    99→    99→
   100→   100→      // Group by agency
   101→   101→      const groupedByAgency = groupByAgency(filteredSets);
   102→   102→      setAgencyGroups(groupedByAgency);
   103→   103→    } catch (error) {
   104→   104→      console.error('Error fetching QR sets:', error);
   105→   105→    } finally {
   106→   106→      setLoading(false);
   107→   107→    }
   108→   108→  };
   109→   109→
   110→   110→  const groupByAgency = (sets: QRSet[]): AgencyGroup[] => {
   111→   111→    const groups = new Map<string | null, AgencyGroup>();
   112→   112→
   113→   113→    sets.forEach(set => {
   114→   114→      const key = set.agencyId || 'no-agency';
   115→   115→      const agencyName = set.agencyName || 'Sans agence';
   116→   116→
   117→   117→      if (!groups.has(key)) {
   118→   118→        groups.set(key, {
   119→   119→          agencyId: set.agencyId,
   120→   120→          agencyName,
   121→   121→          sets: [],
   122→   122→          totalQr: 0,
   123→   123→          isExpanded: true, // Expanded by default
   124→   124→        });
   125→   125→      }
   126→   126→
   127→   127→      const group = groups.get(key)!;
   128→   128→      group.sets.push(set);
   129→   129→      group.totalQr += set.qrCount;
   130→   130→    });
   131→   131→
   132→   132→    // Sort: agencies with name first, then "Sans agence"
   133→   133→    return Array.from(groups.values()).sort((a, b) => {
   134→   134→      if (a.agencyId === null) return 1;
   135→   135→      if (b.agencyId === null) return -1;
   136→   136→      return a.agencyName.localeCompare(b.agencyName);
   137→   137→    });
   138→   138→  };
   139→   139→
   140→   140→  const getActivationStatus = (status: string): 'new' | 'partial' | 'activated' => {
   141→   141→    if (status === 'active' || status === 'scanned') return 'activated';
   142→   142→    if (status === 'partial') return 'partial';
   143→   143→    return 'new';
   144→   144→  };
   145→   145→
   146→   146→  const toggleAgencyGroup = (agencyId: string | null) => {
   147→   147→    setAgencyGroups(prev => 
   148→   148→      prev.map(group => 
   149→   149→        group.agencyId === agencyId 
   150→   150→          ? { ...group, isExpanded: !group.isExpanded }
   151→   151→          : group
   152→   152→      )
   153→   153→    );
   154→   154→  };
   155→   155→
   156→   156→  const handleDeleteSet = async () => {
   157→   157→    if (!selectedSet) return;
   158→   158→
   159→   159→    try {
   160→   160→      const params = new URLSearchParams({ setId: selectedSet.setId });
   161→   161→      const response = await fetch(`/api/qrcodes?${params}`, { method: 'DELETE' });
   162→   162→      const data = await response.json();
   163→   163→
   164→   164→      if (response.ok && data.success) {
   165→   165→        fetchSets();
   166→   166→        setShowDeleteModal(false);
   167→   167→        setSelectedSet(null);
   168→   168→      } else {
   169→   169→        alert(`Erreur: ${data.error || 'Erreur lors de la suppression'}`);
   170→   170→      }
   171→   171→    } catch (error) {
   172→   172→      console.error('Error deleting set:', error);
   173→   173→      alert('Erreur lors de la suppression');
   174→   174→    }
   175→   175→  };
   176→   176→
   177→   177→  const handleDownloadPDF = async (set: QRSet) => {
   178→   178→    setIsDownloading(true);
   179→   179→    setSelectedSet(set);
   180→   180→
   181→   181→    try {
   182→   182→      const canvas = document.createElement('canvas');
   183→   183→      const ctx = canvas.getContext('2d');
   184→   184→      if (!ctx) throw new Error('Canvas not supported');
   185→   185→
   186→   186→      const qrSize = 300;
   187→   187→      const padding = 40;
   188→   188→      const headerHeight = 100;
   189→   189→
   190→   190→      const cols = Math.min(set.qrCount, 3);
   191→   191→      const rows = Math.ceil(set.qrCount / 3);
   192→   192→
   193→   193→      canvas.width = cols * qrSize + (cols + 1) * padding;
   194→   194→      canvas.height = headerHeight + rows * qrSize + (rows + 1) * padding + 60;
   195→   195→
   196→   196→      // Background
   197→   197→      ctx.fillStyle = '#ffffff';
   198→   198→      ctx.fillRect(0, 0, canvas.width, canvas.height);
   199→   199→
   200→   200→      // Header
   201→   201→      ctx.fillStyle = set.type === 'hajj' ? '#1D4ED8' : '#f59e0b';
   202→   202→      ctx.fillRect(0, 0, canvas.width, headerHeight);
   203→   203→
   204→   204→      ctx.fillStyle = '#ffffff';
   205→   205→      ctx.font = 'bold 24px Arial';
   206→   206→      ctx.textAlign = 'center';
   207→   207→      ctx.fillText('QRPass - Étiquettes', canvas.width / 2, 35);
   208→   208→
   209→   209→      ctx.font = '16px Arial';
   210→   210→      ctx.fillText(`${set.setId} | Hajj 2026 | ${set.qrCount} QR`, canvas.width / 2, 65);
   211→   211→      if (set.agencyName) {
   212→   212→        ctx.font = '14px Arial';
   213→   213→        ctx.fillText(`Agence: ${set.agencyName}`, canvas.width / 2, 85);
   214→   214→      }
   215→   215→
   216→   216→      // Generate QR codes
   217→   217→      for (let i = 0; i < set.references.length; i++) {
   218→   218→        const col = i % 3;
   219→   219→        const row = Math.floor(i / 3);
   220→   220→        const x = padding + col * (qrSize + padding);
   221→   221→        const y = headerHeight + padding + row * (qrSize + padding);
   222→   222→
   223→   223→        ctx.fillStyle = '#ffffff';
   224→   224→        ctx.fillRect(x, y, qrSize, qrSize);
   225→   225→
   226→   226→        const qrUrl = `${window.location.origin}/found/${set.references[i]}`;
   227→   227→        const svgElement = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
   228→   228→        const qrSvg = <QRCodeSVG value={qrUrl} size={qrSize - 40} level="H" fgColor={set.type === 'hajj' ? '#1D4ED8' : '#f59e0b'} />;
   229→   229→
   230→   230→        // Use a promise-based approach to render SVG to canvas
   231→   231→        const svgString = `<svg xmlns="http://www.w3.org/2000/svg" width="${qrSize - 40}" height="${qrSize - 40}">
   232→   232→          <rect width="100%" height="100%" fill="white"/>
   233→   233→          ${generateQRPath(qrUrl, qrSize - 40, set.type === 'hajj' ? '#1D4ED8' : '#f59e0b')}
   234→   234→        </svg>`;
   235→   235→
   236→   236→        const img = new Image();
   237→   237→        const svgBlob = new Blob([svgString], { type: 'image/svg+xml;charset=utf-8' });
   238→   238→        const url = URL.createObjectURL(svgBlob);
   239→   239→
   240→   240→        await new Promise<void>((resolve) => {
   241→   241→          img.onload = () => {
   242→   242→            ctx.drawImage(img, x + 20, y + 20, qrSize - 40, qrSize - 40);
   243→   243→            URL.revokeObjectURL(url);
   244→   244→            resolve();
   245→   245→          };
   246→   246→          img.onerror = () => {
   247→   247→            URL.revokeObjectURL(url);
   248→   248→            resolve();
   249→   249→          };
   250→   250→          img.src = url;
   251→   251→        });
   252→   252→
   253→   253→        // Reference label
   254→   254→        ctx.fillStyle = '#1e293b';
   255→   255→        ctx.font = 'bold 12px Arial';
   256→   256→        ctx.textAlign = 'center';
   257→   257→        ctx.fillText(set.references[i], x + qrSize / 2, y + qrSize - 15);
   258→   258→      }
   259→   259→
   260→   260→      // Footer
   261→   261→      ctx.fillStyle = '#64748b';
   262→   262→      ctx.font = '12px Arial';
   263→   263→      ctx.textAlign = 'center';
   264→   264→      ctx.fillText(`Généré le ${new Date().toLocaleDateString('fr-FR')} | QRPass.com`, canvas.width / 2, canvas.height - 20);
   265→   265→
   266→   266→      // Download
   267→   267→      const link = document.createElement('a');
   268→   268→      link.download = `QRPass-${set.setId}.png`;
   269→   269→      link.href = canvas.toDataURL('image/png');
   270→   270→      link.click();
   271→   271→
   272→   272→      setIsDownloading(false);
   273→   273→    } catch (error) {
   274→   274→      console.error('Error generating PDF:', error);
   275→   275→      setIsDownloading(false);
   276→   276→      alert('Erreur lors de la génération');
   277→   277→    }
   278→   278→  };
   279→   279→
   280→   280→  // Simple QR path generator fallback
   281→   281→  const generateQRPath = (url: string, size: number, color: string): string => {
   282→   282→    return `<text x="50%" y="50%" text-anchor="middle" fill="${color}" font-size="10">${url}</text>`;
   283→   283→  };
   284→   284→
   285→   285→  const handleShareSet = async (set: QRSet) => {
   286→   286→    const shareUrl = `${window.location.origin}/found/${set.references[0]}`;
   287→   287→    if (navigator.share) {
   288→   288→      try {
   289→   289→        await navigator.share({
   290→   290→          title: `QRPass - ${set.setId}`,
   291→   291→          text: `${set.qrCount} QR codes pour ${set.agencyName || 'agence'}`,
   292→   292→          url: shareUrl,
   293→   293→        });
   294→   294→      } catch (err) {
   295→   295→        console.log('Share cancelled');
   296→   296→      }
   297→   297→    } else {
   298→   298→      navigator.clipboard.writeText(shareUrl);
   299→   299→      alert('Lien copié !');
   300→   300→    }
   301→   301→  };
   302→   302→
   303→   303→  const formatDate = (date: Date | string) => {
   304→   304→    return new Date(date).toLocaleDateString('fr-FR', {
   305→   305→      day: '2-digit',
   306→   306→      month: 'short',
   307→   307→      year: 'numeric',
   308→   308→    });
   309→   309→  };
   310→   310→
   311→   311→  return (
   312→   312→    <div className="max-w-7xl mx-auto">
   313→   313→      {/* Header */}
   314→   314→      <div className="flex flex-col md:flex-row md:items-center justify-between gap-4 mb-6">
   315→   315→        <div>
   316→   316→          <h1 className="text-2xl font-bold text-slate-800 dark:text-white">Étiquettes QR</h1>
   317→   317→          <p className="text-slate-500 dark:text-slate-400 text-sm">
   318→   318→            {stats.totalSets} sets • {stats.totalQr} QR codes
   319→   319→          </p>
   320→   320→        </div>
   321→   321→
   322→   322→        <div className="flex items-center gap-3">
   323→   323→          <div className="relative">
   324→   324→            <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" />
   325→   325→            <input
   326→   326→              type="text"
   327→   327→              placeholder="Rechercher..."
   328→   328→              value={search}
   329→   329→              onChange={(e) => setSearch(e.target.value)}
   330→   330→              className="pl-10 pr-4 py-2 w-64 bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-xl text-sm focus:outline-none focus:ring-2 focus:ring-green-500"
   331→   331→            />
   332→   332→          </div>
   333→   333→
   334→   334→          <button
   335→   335→            onClick={() => fetchSets()}
   336→   336→            className="p-2 bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-xl hover:bg-slate-50 dark:hover:bg-slate-700 transition-colors"
   337→   337→          >
   338→   338→            <RefreshCw className={`w-4 h-4 text-slate-500 ${loading ? 'animate-spin' : ''}`} />
   339→   339→          </button>
   340→   340→
   341→   341→          <Link
   342→   342→            href="/admin/generer"
   343→   343→            className="flex items-center gap-2 px-4 py-2 bg-green-500 hover:bg-green-600 text-white rounded-xl font-medium transition-colors"
   344→   344→          >
   345→   345→            <QrCode className="w-4 h-4" />
   346→   346→            Générer QR
   347→   347→          </Link>
   348→   348→        </div>
   349→   349→      </div>
   350→   350→
   351→   351→      {/* Tabs */}
   352→   352→      <div className="flex gap-2 mb-6">
   353→   353→        <button
   354→   354→          onClick={() => setActiveTab('hajj')}
   355→   355→          className={`flex items-center gap-2 px-6 py-3 rounded-xl font-medium transition-all ${
   356→   356→            activeTab === 'hajj'
   357→   357→              ? 'bg-blue-700 text-white shadow-lg shadow-blue-700/30'
   358→   358→              : 'bg-white dark:bg-slate-800 text-slate-600 dark:text-slate-300 border border-slate-200 dark:border-slate-700 hover:bg-slate-50 dark:hover:bg-slate-700'
   359→   359→          }`}
   360→   360→        >
   361→   361→          <Luggage className="w-5 h-5" />
   362→   362→          Hajj
   363→   363→          <span className={`px-2 py-0.5 rounded-full text-xs ${
   364→   364→            activeTab === 'hajj' 
   365→   365→              ? 'bg-white/20 text-white' 
   366→   366→              : 'bg-slate-100 dark:bg-slate-700 text-slate-600 dark:text-slate-400'
   367→   367→          }`}>
   368→   368→            {stats.hajjSets}
   369→   369→          </span>
   370→   370→        </button>
   371→   371→      </div>
   372→   372→
   373→   373→      {/* Content */}
   374→   374→      {loading ? (
   375→   375→        <div className="flex items-center justify-center py-20">
   376→   376→          <div className="animate-spin w-8 h-8 border-4 border-slate-200 border-t-green-500 rounded-full" />
   377→   377→        </div>
   378→   378→      ) : agencyGroups.length === 0 ? (
   379→   379→        <div className="bg-white dark:bg-slate-800 rounded-2xl border border-slate-200 dark:border-slate-700 p-12 text-center">
   380→   380→          <div className={`w-16 h-16 mx-auto mb-4 rounded-2xl flex items-center justify-center ${
   381→   381→            activeTab === 'hajj' ? 'bg-emerald-100 dark:bg-blue-600/20' : 'bg-amber-100 dark:bg-blue-600/20'
   382→   382→          }`}>
   383→   383→            {activeTab === 'hajj' ? (
   384→   384→              <Luggage className="w-8 h-8 text-blue-700" />
   385→   385→            ) : (
   386→   386→              <Plane className="w-8 h-8 text-amber-600" />
   387→   387→            )}
   388→   388→          </div>
   389→   389→          <h3 className="text-lg font-semibold text-slate-800 dark:text-white mb-2">
   390→   390→            Aucun QR code Hajj
   391→   391→          </h3>
   392→   392→          <p className="text-slate-500 dark:text-slate-400 mb-4">
   393→   393→            {search ? 'Aucun résultat pour votre recherche' : 'Commencez par générer des QR codes'}
   394→   394→          </p>
   395→   395→          {!search && (
   396→   396→            <Link
   397→   397→              href="/admin/generer"
   398→   398→              className="inline-flex items-center gap-2 px-4 py-2 bg-green-500 hover:bg-green-600 text-white rounded-xl font-medium transition-colors"
   399→   399→            >
   400→   400→              <QrCode className="w-4 h-4" />
   401→   401→              Générer des QR codes

   404→   404→        </div>
   405→   405→      ) : (

   411→   411→            >
   412→   412→              {/* Agency Header */}
   413→   413→              <button

   416→   416→              >

   439→   439→              </button>

   442→   442→              {group.isExpanded && (
   443→   443→                <div className="border-t border-slate-200 dark:border-slate-700">
   444→   444→                  <div className="divide-y divide-slate-100 dark:divide-slate-700">
