    25→import { useAgency } from '../layout';

    64→
    65→export default function BaggagesPage() {
    66→  const { agencyId, agencyName } = useAgency();
    67→  const [baggages, setBaggages] = useState<Baggage[]>([]);
    68→  const [filteredBaggages, setFilteredBaggages] = useState<Baggage[]>([]);
    69→  const [loading, setLoading] = useState(true);
    70→  const [search, setSearch] = useState('');
    71→  const [statusFilter, setStatusFilter] = useState('all');
    72→  const [selectedBaggage, setSelectedBaggage] = useState<Baggage | null>(null);
    73→  const [showDetailModal, setShowDetailModal] = useState(false);
    74→  const [actionLoading, setActionLoading] = useState<string | null>(null);
    75→
    76→  // Bulk selection state
    77→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
    78→  const [deleteLoading, setDeleteLoading] = useState(false);
    79→  const [showDeleteConfirm, setShowDeleteConfirm] = useState(false);
    80→
    81→  const [pilgrims, setPilgrims] = useState<Pilgrim[]>([]);
    82→  const [pilgrimsLoading, setPilgrimsLoading] = useState(true);
    83→  const [selectedPilgrim, setSelectedPilgrim] = useState<Pilgrim | null>(null);
    84→  const [showPilgrimDetail, setShowPilgrimDetail] = useState(false);
    85→
    86→  useEffect(() => {
    87→    fetchBaggages();
    88→    fetchPilgrims();
    89→  }, [agencyId]);
    90→
    91→  useEffect(() => {
    92→    filterBaggages();
    93→  }, [baggages, search, statusFilter]);
    94→
    95→  const fetchBaggages = async () => {
    96→    try {
    97→      const params = new URLSearchParams({
    98→        agencyId: agencyId,
    99→      });
   100→
   101→      const response = await fetch(`/api/agency/baggages?${params}`);
   102→      const data = await response.json();
   103→      setBaggages(data.baggages || []);
   104→    } catch (error) {
   105→      console.error('Error fetching baggages:', error);
   106→    } finally {
   107→      setLoading(false);
   108→    }
   109→  };
   110→
   111→  const fetchPilgrims = async () => {
   112→    try {
   113→      const params = new URLSearchParams({ agencyId });
   114→      const response = await fetch(`/api/agency/pilgrims?${params}`);
   115→      const data = await response.json();
   116→      setPilgrims(data.pilgrims || []);
   117→    } catch (error) {
   118→      console.error('Error fetching pilgrims:', error);
   119→    } finally {
   120→      setPilgrimsLoading(false);
   121→    }
   122→  };
   123→
   124→  const filterBaggages = () => {
   125→    let filtered = [...baggages];
   126→
   127→    if (statusFilter !== 'all') {
   128→      filtered = filtered.filter(b => b.status === statusFilter);
   129→    }
   130→
   131→    if (search.trim()) {
   132→      const searchLower = search.toLowerCase();
   133→      filtered = filtered.filter(b =>
   134→        b.reference.toLowerCase().includes(searchLower) ||
   135→        `${b.travelerFirstName || ''} ${b.travelerLastName || ''}`.toLowerCase().includes(searchLower)
   136→      );
   137→    }
   138→
   139→    setFilteredBaggages(filtered);
   140→  };
   141→
   142→  // AGENCY-FIX: Split filtered baggages into activated and pending sections
   143→  const activatedBaggages = filteredBaggages.filter(b =>
   144→    isActive(b.status) || b.travelerFirstName !== null || b.status === 'lost' || b.status === 'found' || b.status === 'blocked'
   145→  );
   146→  const pendingBaggages = filteredBaggages.filter(b =>
   147→    isPending(b.status) && b.travelerFirstName === null && b.travelerLastName === null
   148→  );
   149→
   150→  // Bulk selection handlers
   151→  const toggleSelect = (id: string) => {
   152→    setSelectedIds(prev => {
   153→      const next = new Set(prev);
   154→      if (next.has(id)) {
   155→        next.delete(id);
   156→      } else {
   157→        next.add(id);
   158→      }
   159→      return next;
   160→    });
   161→  };
   162→
   163→  const toggleSelectAll = (baggageList: Baggage[]) => {
   164→    setSelectedIds(prev => {
   165→      const next = new Set(prev);
   166→      const allSelected = baggageList.every(b => next.has(b.id));
   167→      if (allSelected) {
   168→        // Deselect all in this list
   169→        for (const b of baggageList) {
   170→          next.delete(b.id);
   171→        }
   172→      } else {
   173→        // Select all in this list
   174→        for (const b of baggageList) {
   175→          next.add(b.id);
   176→        }
   177→      }
   178→      return next;
   179→    });
   180→  };
   181→
   182→  const handleBulkDelete = async () => {
   183→    if (selectedIds.size === 0) return;
   184→    setShowDeleteConfirm(true);
   185→  };
   186→
   187→  const confirmBulkDelete = async () => {
   188→    setDeleteLoading(true);
   189→    try {
   190→      const response = await fetch('/api/agency/baggages', {
   191→        method: 'DELETE',
   192→        headers: { 'Content-Type': 'application/json' },
   193→        body: JSON.stringify({
   194→          agencyId,
   195→          ids: Array.from(selectedIds),
   196→        }),
   197→      });
   198→
   199→      const data = await response.json();
   200→
   201→      if (response.ok) {
   202→        // Remove deleted items from local state
   203→        const deletedSet = new Set(selectedIds);
   204→        setBaggages(prev => prev.filter(b => !deletedSet.has(b.id)));
   205→        setSelectedIds(new Set());
   206→        setShowDeleteConfirm(false);
   207→        alert(`${data.deleted} QR code(s) supprimé(s) avec succès`);
   208→      } else {
   209→        alert(data.error || 'Erreur lors de la suppression');
   210→      }
   211→    } catch (error) {
   212→      console.error('Bulk delete error:', error);
   213→      alert('Erreur lors de la suppression');
   214→    } finally {
   215→      setDeleteLoading(false);
   216→    }
   217→  };
   218→
   219→  const handleDeleteAllPending = async () => {
   220→    if (!confirm(`Supprimer TOUS les QR codes en attente d'activation de votre agence ?\n\nCette action est irréversible.`)) return;
   221→
   222→    setDeleteLoading(true);
   223→    try {
   224→      const response = await fetch('/api/agency/baggages', {
   225→        method: 'DELETE',
   226→        headers: { 'Content-Type': 'application/json' },
   227→        body: JSON.stringify({
   228→          agencyId,
   229→          status: 'pending_activation',
   230→        }),
   231→      });
   232→
   233→      const data = await response.json();
   234→
   235→      if (response.ok) {
   236→        setBaggages(prev => prev.filter(b => !isPending(b.status)));
   237→        setSelectedIds(new Set());
   238→        alert(`${data.deleted} QR code(s) en attente supprimé(s)`);
   239→      } else {
   240→        alert(data.error || 'Erreur lors de la suppression');
   241→      }
   242→    } catch (error) {
   243→      console.error('Delete all pending error:', error);
   244→      alert('Erreur lors de la suppression');
   245→    } finally {
   246→      setDeleteLoading(false);
   247→    }
   248→  };
   249→
   250→  const formatDate = (dateString: string | null) => {
   251→    if (!dateString) return 'Jamais';
   252→    const date = new Date(dateString);
   253→    return date.toLocaleDateString('fr-FR', {
   254→      day: '2-digit',
   255→      month: '2-digit',
   256→      year: 'numeric'
   257→    });
   258→  };
   259→
   260→  const formatDateTime = (dateString: string | null) => {
   261→    if (!dateString) return '-';
   262→    const date = new Date(dateString);
   263→    return date.toLocaleDateString('fr-FR', {
   264→      day: '2-digit',
   265→      month: '2-digit',
   266→      year: 'numeric'
   267→    }) + ' à ' + date.toLocaleTimeString('fr-FR', {
   268→      hour: '2-digit',
   269→      minute: '2-digit'
   270→    });
   271→  };
   272→
   273→  // Handle Declare Lost
   274→  const handleDeclareLost = async (baggageId: string) => {
   275→    if (!confirm('⚠️ Êtes-vous sûr de vouloir déclarer ce bagage comme perdu ?\n\nUne alerte sera envoyée au SuperAdmin.')) return;
   276→
   277→    setActionLoading(baggageId);
   278→    try {
   279→      const response = await fetch(`/api/baggage/${baggageId}/declare-lost`, {
   280→        method: 'PUT',
   281→      });
   282→      const data = await response.json();
   283→
   284→      if (response.ok) {
   285→        setBaggages(prev => prev.map(b => 
   286→          b.id === baggageId ? { ...b, status: 'lost' } : b
   287→        ));
   288→        setShowDetailModal(false);
   289→        setSelectedBaggage(null);
   290→      } else {
   291→        alert(data.error || 'Erreur lors de la déclaration');
   292→      }
   293→    } catch (error) {
   294→      console.error('Declare lost error:', error);
   295→      alert('Erreur lors de la déclaration');
   296→    } finally {
   297→      setActionLoading(null);
   298→    }
   299→  };
   300→
   301→  // Handle Mark Found
   302→  const handleMarkFound = async (baggageId: string) => {
   303→    if (!confirm('✅ Marquer ce bagage comme retrouvé ?')) return;
   304→
   305→    setActionLoading(baggageId);
   306→    try {
   307→      const response = await fetch(`/api/baggage/${baggageId}/mark-found`, {
   308→        method: 'PUT',
   309→      });
   310→      const data = await response.json();
   311→
   312→      if (response.ok) {
   313→        setBaggages(prev => prev.map(b => 
   314→          b.id === baggageId ? { ...b, status: 'found' } : b
   315→        ));
   316→        setShowDetailModal(false);
   317→        setSelectedBaggage(null);
   318→      } else {
   319→        alert(data.error || 'Erreur lors de la mise à jour');
   320→      }
   321→    } catch (error) {
   322→      console.error('Mark found error:', error);
   323→      alert('Erreur lors de la mise à jour');
   324→    } finally {
   325→      setActionLoading(null);
   326→    }
   327→  };
   328→
   329→  const getStatusBadge = (status: string) => {
   330→    const statusConfig: Record<string, { label: string; className: string }> = {
   331→      pending_activation: { label: 'En attente', className: 'bg-amber-100 dark:bg-blue-600/10 text-amber-700 dark:text-blue-500' },
   332→      active: { label: 'Actif', className: 'bg-emerald-100 dark:bg-blue-600/10 text-emerald-700 dark:text-blue-500' },
   333→      scanned: { label: 'Scanné', className: 'bg-blue-100 dark:bg-blue-500/10 text-blue-700 dark:text-blue-400' },
   334→      lost: { label: 'Perdu', className: 'bg-rose-100 dark:bg-rose-500/10 text-rose-700 dark:text-rose-400' },
   335→      found: { label: 'Retrouvé', className: 'bg-green-100 dark:bg-green-500/10 text-green-700 dark:text-green-400' },
   336→      blocked: { label: 'Bloqué', className: 'bg-slate-100 dark:bg-slate-500/10 text-slate-600 dark:text-slate-400' },
   337→    };
   338→
   339→    const config = statusConfig[status] || { label: status, className: 'bg-slate-100 dark:bg-slate-500/10 text-slate-600 dark:text-slate-400' };
   340→
   341→    return (
   342→      <span className={`px-3 py-1 rounded-full text-xs font-medium ${config.className}`}>
   343→        {config.label}
   344→      </span>
   345→    );
   346→  };
   347→
   348→  const filterButtons = [
   349→    { id: 'all', label: 'Tous' },
   350→    { id: 'scanned', label: 'Scannés' },
   351→    { id: 'pending_activation', label: 'En attente' },
   352→    { id: 'lost', label: 'Perdus' },
   353→    { id: 'found', label: 'Retrouvés' },
   354→  ];
   355→
   356→  return (
   357→    <div className="max-w-7xl mx-auto">
   358→      {/* Header */}
   359→      <div className="mb-8 flex items-center justify-between flex-wrap gap-4">
   360→        <div>
   361→          <h1 className="text-2xl font-bold text-slate-800 dark:text-white">Gestion des bagages</h1>
   362→          <p className="text-slate-500 dark:text-slate-400 mt-1">Liste complète des bagages de votre agence</p>
   363→        </div>
   364→        {/* Delete All Pending Button */}
   365→        {pendingBaggages.length > 0 && selectedIds.size === 0 && (
   366→          <button
   367→            onClick={handleDeleteAllPending}
   368→            disabled={deleteLoading}
   369→            className="inline-flex items-center gap-2 px-4 py-2.5 bg-rose-500 hover:bg-rose-600 text-white rounded-xl text-sm font-medium transition-colors shadow-lg shadow-rose-500/20 disabled:opacity-50"
   370→          >
   371→            {deleteLoading ? (
   372→              <Loader2 className="w-4 h-4 animate-spin" />
   373→            ) : (
   374→              <Trash2 className="w-4 h-4" />
   375→            )}
   376→            Supprimer tous les QR en attente ({pendingBaggages.length})
   377→          </button>
   378→        )}
   379→      </div>
   380→
   381→      {/* Bulk Action Bar - Shows when items are selected */}
   382→      {selectedIds.size > 0 && (
   383→        <div className="mb-6 bg-rose-50 dark:bg-rose-500/10 border border-rose-200 dark:border-rose-800 rounded-xl px-5 py-3 flex items-center justify-between flex-wrap gap-3">
   384→          <div className="flex items-center gap-3">
   385→            <div className="w-8 h-8 rounded-lg bg-rose-500 flex items-center justify-center">
   386→              <Trash2 className="w-4 h-4 text-white" />
   387→            </div>
   388→            <span className="text-rose-700 dark:text-rose-400 font-medium text-sm">
   389→              {selectedIds.size} QR code(s) sélectionné(s)
   390→            </span>
   391→          </div>
   392→          <div className="flex items-center gap-2">
   393→            <button
   394→              onClick={() => setSelectedIds(new Set())}
   395→              className="px-4 py-2 text-slate-600 dark:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-800 rounded-lg text-sm transition-colors"
   396→            >
   397→              Annuler
   398→            </button>
   399→            <button
   400→              onClick={handleBulkDelete}
   401→              disabled={deleteLoading}
   402→              className="inline-flex items-center gap-2 px-4 py-2 bg-rose-500 hover:bg-rose-600 text-white rounded-lg text-sm font-medium transition-colors disabled:opacity-50"
   403→            >
   404→              {deleteLoading ? (
   405→                <Loader2 className="w-4 h-4 animate-spin" />
   406→              ) : (
   407→                <Trash2 className="w-4 h-4" />
   408→              )}
   409→              Supprimer la sélection
   410→            </button>
   411→          </div>
   412→        </div>
   413→      )}
   414→
   415→      {/* Stats Cards - Multicolored */}
   416→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mb-8">
   417→        <div className="kpi-card kpi-card-green p-5">
   418→          <div className="w-10 h-10 rounded-xl bg-white/20 flex items-center justify-center mb-3">
   419→            <Luggage className="w-5 h-5 text-white" />
   420→          </div>
   421→          <p className="text-2xl font-bold text-white">{baggages.length}</p>
   422→          <p className="text-sm text-white/80">Total bagages</p>
   423→        </div>
   424→        <div className="kpi-card kpi-card-blue p-5">
   425→          <div className="w-10 h-10 rounded-xl bg-white/20 flex items-center justify-center mb-3">
   426→            <CheckCircle className="w-5 h-5 text-white" />
   427→          </div>
   428→          <p className="text-2xl font-bold text-white">{baggages.filter(b => isActive(b.status)).length}</p>
   429→          <p className="text-sm text-white/80">Actifs</p>
   430→        </div>
   431→        <div className="kpi-card kpi-card-orange p-5">
   432→          <div className="w-10 h-10 rounded-xl bg-white/20 flex items-center justify-center mb-3">
   433→            <Clock className="w-5 h-5 text-white" />
   434→          </div>
   435→          <p className="text-2xl font-bold text-white">{baggages.filter(b => isPending(b.status)).length}</p>
   436→          <p className="text-sm text-white/80">En attente</p>
   437→        </div>
   438→        <div className="kpi-card kpi-card-red p-5">
   439→          <div className="w-10 h-10 rounded-xl bg-white/20 flex items-center justify-center mb-3">
   440→            <AlertTriangle className="w-5 h-5 text-white" />
   441→          </div>
   442→          <p className="text-2xl font-bold text-white">{baggages.filter(b => b.status === 'lost').length}</p>
   443→          <p className="text-sm text-white/80">Perdus</p>
   444→        </div>
   445→      </div>
   446→
   447→      {/* Search Bar */}
   448→      <div className="mb-6">
   449→        <div className="relative">
   450→          <Search className="absolute left-4 top-1/2 -translate-y-1/2 w-5 h-5 text-slate-400" />
   451→          <input
   452→            type="text"
   453→            placeholder="Rechercher par nom ou référence..."
   454→            value={search}
   455→            onChange={(e) => setSearch(e.target.value)}
   456→            className="w-full bg-white dark:bg-slate-900 border border-slate-200 dark:border-slate-800 rounded-xl py-3 pl-12 pr-4 text-slate-700 dark:text-slate-200 placeholder-slate-400 focus:ring-2 focus:ring-blue-600/20 focus:border-blue-600 transition-all"
   457→          />
   458→        </div>
   459→      </div>
   460→
   461→      {/* Filter Buttons */}
   462→      <div className="flex flex-wrap gap-2 mb-6">
   463→        {filterButtons.map((btn) => (
   464→          <button
   465→            key={btn.id}
   466→            onClick={() => setStatusFilter(btn.id)}
   467→            className={`px-4 py-2 rounded-xl text-sm font-medium transition-all duration-200 ${
   468→              statusFilter === btn.id
   469→                ? 'bg-blue-600 text-white shadow-lg'
   470→                : 'bg-white dark:bg-slate-900 text-slate-600 dark:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-800 border border-slate-200 dark:border-slate-800'
   471→            }`}
   472→          >
   473→            {btn.label}
   474→          </button>
   475→        ))}
   476→      </div>
   477→
   478→      {/* Loading state */}
   479→      {loading ? (
   480→        <div className="bg-white dark:bg-slate-900 rounded-2xl border border-slate-200 dark:border-slate-800 overflow-hidden">
   481→          <div className="text-center py-12">
   482→            <div className="flex items-center justify-center gap-3">
   483→              <div className="w-6 h-6 border-2 border-blue-600/30 border-t-blue-600 rounded-full animate-spin" />
   484→              <span className="text-slate-500">Chargement...</span>
   485→            </div>
   486→          </div>
   487→        </div>
   488→      ) : filteredBaggages.length === 0 ? (
   489→        <div className="bg-white dark:bg-slate-900 rounded-2xl border border-slate-200 dark:border-slate-800 overflow-hidden">
   490→          <div className="text-center py-12">
   491→            <div className="flex flex-col items-center">
   492→              <div className="w-16 h-16 bg-slate-100 dark:bg-slate-800 rounded-full flex items-center justify-center mb-4">
   493→                <Luggage className="w-8 h-8 text-slate-400" />
   494→              </div>
   495→              <p className="text-slate-500 dark:text-slate-400">Aucun bagage trouvé</p>
   496→            </div>
   497→          </div>
   498→        </div>
   499→      ) : (
   500→        <>
   501→          {/* Section 1 — Bagages activés */}
   502→          {activatedBaggages.length > 0 && (
   503→            <div className="bg-white dark:bg-slate-900 rounded-2xl border border-slate-200 dark:border-slate-800 overflow-hidden mb-6">
   504→              <div className="px-6 py-4 border-b border-slate-200 dark:border-slate-800 bg-emerald-50/50 dark:bg-blue-600/5">
   505→                <div className="flex items-center gap-2">
   506→                  <div className="w-3 h-3 rounded-full bg-blue-600" />
   507→                  <h2 className="text-sm font-semibold text-slate-800 dark:text-white">
   508→                    Bagages activés ({activatedBaggages.length})
   509→                  </h2>
   510→                </div>
   511→              </div>
   512→              <div className="overflow-x-auto">
   513→                <table className="w-full">
   514→                  <thead>
   515→                    <tr className="border-b border-slate-200 dark:border-slate-800 bg-slate-50 dark:bg-slate-800/50">
   516→                      <th className="text-left px-4 py-4 w-10">
   517→                        <input
   518→                          type="checkbox"
   519→                          checked={activatedBaggages.length > 0 && activatedBaggages.every(b => selectedIds.has(b.id))}
   520→                          onChange={() => toggleSelectAll(activatedBaggages)}
   521→                          className="w-4 h-4 rounded border-slate-300 text-rose-500 focus:ring-rose-500 cursor-pointer"
