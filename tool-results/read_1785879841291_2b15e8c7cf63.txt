   294→   294→// ─── Helper: Parse hotelCoords ───
   295→   295→function parseCoords(coords: { lat: number; lng: number } | string | null): { lat: number; lng: number } | null {
   296→   296→  if (!coords) return null;
   297→   297→  if (typeof coords === 'object' && 'lat' in coords && 'lng' in coords) {
   298→   298→    return { lat: coords.lat, lng: coords.lng };
   299→   299→  }
   300→   300→  if (typeof coords === 'string') {
   301→   301→    try {
   302→   302→      const parsed = JSON.parse(coords);
   303→   303→      if (parsed && typeof parsed === 'object' && 'lat' in parsed && 'lng' in parsed) {
   304→   304→        return { lat: parsed.lat, lng: parsed.lng };
   305→   305→      }
   306→   306→    } catch {
   307→   307→      const parts = coords.split(',').map((s) => parseFloat(s.trim()));
   308→   308→      if (parts.length === 2 && !isNaN(parts[0]) && !isNaN(parts[1])) {
   309→   309→        return { lat: parts[0], lng: parts[1] };
   310→   310→      }
   311→   311→    }
   312→   312→  }
   313→   313→  return null;
   314→   314→}
