   213→// ─── Main Component ───
   214→function BaggageActivateContent() {
   215→  const router = useRouter();
   216→  const searchParams = useSearchParams();
   217→  const qrFromUrl = searchParams.get('code') || searchParams.get('qr') || '';
   218→
   219→  // Language
   220→  const [lang, setLang] = useState<Lang>('fr');
   221→  const dir = lang === 'ar' ? 'rtl' : 'ltr';
   222→  const t = T[lang];
   223→
   224→  // Form state
   225→  const [reference, setReference] = useState('');
   226→  const [fullName, setFullName] = useState('');
   227→  const [phone, setPhone] = useState('');
   228→  const [phoneCountry, setPhoneCountry] = useState('SN'); // default Senegal
   229→  const [email, setEmail] = useState('');
   230→  const [airline, setAirline] = useState('');
   231→  const [flightNumber, setFlightNumber] = useState('');
   232→  const [destination, setDestination] = useState('');
   233→  const [departureDate, setDepartureDate] = useState('');
   234→  const [departureTime, setDepartureTime] = useState('');
   235→  const [hotelName, setHotelName] = useState('');
   236→  const [hotelAddress, setHotelAddress] = useState('');
   237→  const [hotelPhone, setHotelPhone] = useState('');
   238→  const [roomNumber, setRoomNumber] = useState('');
   239→  const [photoPreview, setPhotoPreview] = useState<string | null>(null);
   240→  const [photoFile, setPhotoFile] = useState<File | null>(null);
   241→  const [loading, setLoading] = useState(false);
   242→
   243→  // Deferred activation
   244→  const [deferredActivation, setDeferredActivation] = useState(false);
   245→  const [activationDate, setActivationDate] = useState('');
   246→
   247→  // Pre-fill QR from URL
   248→  useEffect(() => {
   249→    if (qrFromUrl) {
   250→      setReference(qrFromUrl.toUpperCase());
   251→    }
   252→  }, [qrFromUrl]);
   253→
   254→  // Detect country from IP for phone input
   255→  useEffect(() => {
   256→    fetch('/api/ip-country')
   257→      .then(res => res.json())
   258→      .then(data => {
   259→        if (data.country) setPhoneCountry(data.country);
   260→      })
   261→      .catch(() => {}); // silent fallback to SN
   262→  }, []);
   263→
   264→  // Photo preview
   265→  const handlePhotoChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   266→    const file = e.target.files?.[0];
   267→    if (file) {
   268→      setPhotoFile(file);
   269→      const reader = new FileReader();
   270→      reader.onload = (ev) => setPhotoPreview(ev.target?.result as string);
   271→      reader.readAsDataURL(file);
   272→    }
   273→  };
   274→
   275→  // Form validation
   276→  const validate = (): boolean => {
   277→    if (!reference.trim()) {
   278→      toast({ title: 'Code QR requis', variant: 'destructive' });
   279→      return false;
   280→    }
   281→    if (!fullName.trim()) {
   282→      toast({ title: 'Nom complet requis', variant: 'destructive' });
   283→      return false;
   284→    }
   285→    if (!phone.trim()) {
   286→      toast({ title: 'Numéro WhatsApp requis', variant: 'destructive' });
   287→      return false;
   288→    }
   289→    if (!airline) {
   290→      toast({ title: 'Compagnie aérienne requise', variant: 'destructive' });
   291→      return false;
   292→    }
   293→    if (!flightNumber.trim()) {
   294→      toast({ title: 'Numéro de vol requis', variant: 'destructive' });
   295→      return false;
   296→    }
   297→    if (!destination) {
   298→      toast({ title: 'Destination requise', variant: 'destructive' });
   299→      return false;
   300→    }
   301→    if (!departureDate) {
   302→      toast({ title: 'Date de départ requise', variant: 'destructive' });
   303→      return false;
   304→    }
   305→    return true;
   306→  };
   307→
   308→  // Submit
   309→  const handleSubmit = async (e: React.FormEvent) => {
   310→    e.preventDefault();
   311→    if (!validate()) return;
   312→
   313→    setLoading(true);
   314→    try {
   315→      // Split fullName into firstName and lastName for API compatibility
   316→      const nameParts = fullName.trim().split(/\s+/);
   317→      const firstName = nameParts[0] || '';
   318→      const lastName = nameParts.slice(1).join(' ') || firstName;
   319→
   320→      // Upload photo first if selected
   321→      let photoUrl: string | null = null;
   322→      if (photoFile) {
   323→        try {
   324→          const photoFormData = new FormData();
   325→          photoFormData.append('file', photoFile);
   326→          const uploadRes = await fetch('/api/baggage/upload-photo', {
   327→            method: 'POST',
   328→            body: photoFormData,
   329→          });
   330→          if (uploadRes.ok) {
   331→            const uploadData = await uploadRes.json();
   332→            photoUrl = uploadData.photoUrl;
   333→          }
   334→        } catch {
   335→          // Non-blocking, continue without photo
   336→        }
   337→      }
   338→
   339→      const response = await fetch('/api/activate', {
   340→        method: 'POST',
   341→        headers: { 'Content-Type': 'application/json' },
   342→        body: JSON.stringify({
   343→          reference: reference.toUpperCase(),
   344→          travelerFirstName: firstName,
   345→          travelerLastName: lastName,
   346→          whatsappOwner: phone.trim(),
   347→          airlineName: airline,
   348→          flightNumber: flightNumber.trim().toUpperCase(),
   349→          destination,
   350→          hotelName: hotelName.trim() || undefined,
   351→          hotelAddress: hotelAddress.trim() || undefined,
   352→          hotelPhone: hotelPhone.trim() || undefined,
   353→          roomNumber: roomNumber.trim() || undefined,
   354→          departureDate,
   355→          departureTime: departureTime || undefined,
   356→          transportMode: 'flight',
   357→          photoUrl: photoUrl || undefined,
   358→          activationDate: deferredActivation && activationDate ? activationDate : undefined,
   359→        }),
   360→      });
   361→
   362→      if (response.ok) {
   363→        const data = await response.json();
   364→        sessionStorage.setItem('activationData', JSON.stringify({
   365→          reference: reference.toUpperCase(),
   366→          firstName,
   367→          lastName,
   368→          whatsapp: phone.trim(),
   369→          airlineName: airline,
   370→          flightNumber,
   371→          destination,
   372→          hotelName,
   373→          hotelAddress,
   374→          hotelPhone,
   375→          roomNumber,
   376→          type: 'hajj',
   377→          activatedAt: new Date().toISOString(),
   378→          expiresAt: data.baggage?.expiresAt,
   379→          photoUrl: photoUrl || data.baggage?.photoUrl || null,
   380→          // Store base64 preview for instant display on confirmation page (no server request needed)
   381→          photoPreview: photoPreview || null,
   382→        }));
   383→
   384→        const params = new URLSearchParams({
   385→          code: reference.toUpperCase(),
   386→          firstName,
   387→          lastName,
   388→          flight: flightNumber,
   389→          destination,
   390→          chefPhone: phone.trim(),
   391→        });
   392→        if (photoUrl) params.set('photo', photoUrl);
   393→        router.push(`/activate/confirmation?${params.toString()}`);
   394→      } else {
   395→        const error = await response.json();
   396→        toast({ title: error.message || t.errorTitle, variant: 'destructive' });
   397→      }
   398→    } catch {
   399→      toast({ title: t.errorMsg, variant: 'destructive' });
   400→    } finally {
   401→      setLoading(false);
   402→    }
   403→  };
   404→
   405→  // Min date = today
   406→  const today = new Date().toISOString().split('T')[0];
   407→
   408→  return (
   409→    <main dir={dir} className="min-h-screen flex flex-col items-center" style={{ background: BG, color: TEXT, padding: '16px' }}>
   410→
   411→      {/* ─── Header ─── */}
   412→      <div className="w-full max-w-[480px] flex items-center justify-between mb-6 animate-[fadeInDown_0.5s_ease]">
   413→        <div>
   414→          <div className="text-[28px] font-extrabold tracking-tight text-black">
   415→            <Image src="/logo.png" alt="PassHajj" width={120} height={46} style={{ objectFit: 'contain' }} />
   416→          </div>
   417→          <div className="text-sm mt-1 text-black/60">{t.subtitle}</div>
   418→        </div>
   419→        <LangSelector lang={lang} setLang={setLang} />
   420→      </div>
   421→
   422→      {/* ─── Progress Bar ─── */}
   423→      <div className="w-full max-w-[480px] mb-5">
   424→        <div className="h-1 bg-white/40 rounded-full overflow-hidden">
   425→          <div className="h-full bg-black rounded-full" style={{ width: '50%' }} />
   426→        </div>
   427→      </div>
   428→
   429→      {/* ─── Form Card ─── */}
   430→      <form onSubmit={handleSubmit} className="w-full max-w-[480px]">
   431→        <div
   432→          className="rounded-[20px] p-7 shadow-[0_8px_24px_rgba(0,0,0,0.12)] mb-4 animate-[fadeInUp_0.5s_ease_0.2s_both]"
   433→          style={{ background: CARD_BG }}
   434→        >
   435→          <div className="text-lg font-bold mb-5 flex items-center gap-2">
   436→            {t.title}
   437→          </div>
   438→
   439→          {/* QR Code (Auto-détecté) */}
   440→          <div className="mb-5">
   441→            <Label className="text-[13px] font-semibold mb-1.5 block">{t.qrCode}</Label>
   442→            <Input
   443→              value={reference}
   444→              onChange={(e) => setReference(e.target.value.toUpperCase())}
   445→              readOnly={!!qrFromUrl}
   446→              placeholder={t.qrPlaceholder}
   447→              className={`h-[52px] rounded-xl text-[15px] ${qrFromUrl ? 'bg-[#f1f5f9] cursor-not-allowed text-gray-600' : 'bg-[#f8fafc]'} border-[#e2e8f0]`}
   448→            />
   449→            {qrFromUrl && (
   450→              <p className="text-xs mt-1.5 text-gray-500">{t.qrDetected}</p>
   451→            )}
   452→          </div>
   453→
   454→          {/* Full Name */}
   455→          <div className="mb-5">
   456→            <Label className="text-[13px] font-semibold mb-1.5 block">{t.fullName}</Label>
   457→            <Input

   459→              onChange={(e) => setFullName(e.target.value)}

   469→            <PhoneInput

   472→              value={phone}
   473→              onChange={setPhone}

   480→          {/* Email (optional) */}
