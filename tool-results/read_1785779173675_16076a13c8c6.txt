    97→    bloodType: '',
    98→    medicalInfo: '',

   105→    familyContact: '',

   110→

   112→  useEffect(() => {
   113→    const checkCode = async () => {
   114→      try {
   115→        const response = await fetch(`/api/pilgrims/${code}`);
   116→        if (response.ok) {
   117→          const data = await response.json();
   118→          if (data.active) {
   119→            // Already active → redirect to dashboard
   120→            router.push(`/pilgrim/dashboard/${code}`);
   121→            return;
   122→          }

   124→          setPageState('form');

   126→          setPageState('not_found');
   127→        } else {
   128→          setPageState('not_found');
   129→        }
   130→      } catch (error) {
   131→        console.error('Error checking code:', error);
   132→        setPageState('not_found');
   133→      }
   134→    };
   135→
   136→    if (code) {
   137→      checkCode();
   138→    }
   139→  }, [code, router]);
   140→
   141→  const handleSubmit = async (e: React.FormEvent) => {
   142→    e.preventDefault();
   143→    setPageState('submitting');
   144→
   145→    try {
   146→      const response = await fetch(`/api/pilgrims/activate/${code}`, {
   147→        method: 'POST',
   148→        headers: { 'Content-Type': 'application/json' },
   149→        body: JSON.stringify({
   150→          fullName: formData.fullName,
   151→          nationality: formData.nationality,
   152→          bloodType: formData.bloodType || undefined,
   153→          medicalInfo: formData.medicalInfo || undefined,
   154→          hotelMecca: formData.hotelMecca || undefined,
   155→          roomMecca: formData.roomMecca || undefined,
   156→          hotelMedina: formData.hotelMedina || undefined,
   157→          roomMedina: formData.roomMedina || undefined,
   158→          groupLeaderPhone: formData.groupLeaderPhone,
   159→          agencyPhone: formData.agencyPhone || undefined,
   160→          familyContact: formData.familyContact || undefined,
   161→          duration: formData.duration,
   162→        }),
   163→      });

   165→      if (response.ok) {
   166→        setPageState('success');

   170→          title: 'Erreur',
   171→          description: errorData.message || t('pilgrim.activate.submit') + ' — erreur',

   179→        title: 'Erreur',
   180→        description: 'Erreur lors de l\'activation',

   189→  };
