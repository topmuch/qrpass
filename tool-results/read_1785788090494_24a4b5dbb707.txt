   212→// ─── Main Component ───
   213→function BaggageActivateContent() {
   214→  const router = useRouter();
   215→  const searchParams = useSearchParams();
   216→  const qrFromUrl = searchParams.get('code') || searchParams.get('qr') || '';
   217→
   218→  // Language
   219→  const [lang, setLang] = useState<Lang>('fr');
   220→  const dir = lang === 'ar' ? 'rtl' : 'ltr';
   221→  const t = T[lang];
   222→
   223→  // Form state
   224→  const [reference, setReference] = useState('');
   225→  const [fullName, setFullName] = useState('');
   226→  const [phone, setPhone] = useState('');
   227→  const [phoneCountry, setPhoneCountry] = useState('SN'); // default Senegal
   228→  const [email, setEmail] = useState('');
   229→  const [airline, setAirline] = useState('');
   230→  const [flightNumber, setFlightNumber] = useState('');
   231→  const [destination, setDestination] = useState('');
   232→  const [departureDate, setDepartureDate] = useState('');
   233→  const [departureTime, setDepartureTime] = useState('');
   234→  const [hotelName, setHotelName] = useState('');
   235→  const [hotelAddress, setHotelAddress] = useState('');
   236→  const [hotelPhone, setHotelPhone] = useState('');
   237→  const [roomNumber, setRoomNumber] = useState('');
   238→  const [photoPreview, setPhotoPreview] = useState<string | null>(null);
   239→  const [photoFile, setPhotoFile] = useState<File | null>(null);
   240→  const [loading, setLoading] = useState(false);
   241→
   242→  // Deferred activation
   243→  const [deferredActivation, setDeferredActivation] = useState(false);
   244→  const [activationDate, setActivationDate] = useState('');
   245→
   246→  // Pre-fill QR from URL
   247→  useEffect(() => {
   248→    if (qrFromUrl) {
   249→      setReference(qrFromUrl.toUpperCase());
   250→    }
   251→  }, [qrFromUrl]);
   252→
   253→  // Detect country from IP for phone input
   254→  useEffect(() => {
   255→    fetch('/api/ip-country')
   256→      .then(res => res.json())
   257→      .then(data => {
   258→        if (data.country) setPhoneCountry(data.country);
   259→      })
   260→      .catch(() => {}); // silent fallback to SN
   261→  }, []);
   262→
   263→  // Photo preview
   264→  const handlePhotoChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   265→    const file = e.target.files?.[0];
   266→    if (file) {
   267→      setPhotoFile(file);
   268→      const reader = new FileReader();
   269→      reader.onload = (ev) => setPhotoPreview(ev.target?.result as string);
   270→      reader.readAsDataURL(file);
   271→    }
   272→  };
   273→
   274→  // Form validation
   275→  const validate = (): boolean => {
   276→    if (!reference.trim()) {
   277→      toast({ title: 'Code QR requis', variant: 'destructive' });
   278→      return false;
   279→    }
   280→    if (!fullName.trim()) {
   281→      toast({ title: 'Nom complet requis', variant: 'destructive' });
   282→      return false;
   283→    }
   284→    if (!phone.trim()) {
   285→      toast({ title: 'Numéro WhatsApp requis', variant: 'destructive' });
   286→      return false;
   287→    }
   288→    if (!airline) {
   289→      toast({ title: 'Compagnie aérienne requise', variant: 'destructive' });
   290→      return false;
   291→    }
   292→    if (!flightNumber.trim()) {
   293→      toast({ title: 'Numéro de vol requis', variant: 'destructive' });
   294→      return false;
   295→    }
   296→    if (!destination) {
   297→      toast({ title: 'Destination requise', variant: 'destructive' });
   298→      return false;
   299→    }
   300→    if (!departureDate) {
   301→      toast({ title: 'Date de départ requise', variant: 'destructive' });
   302→      return false;
   303→    }
   304→    return true;
   305→  };
   306→
   307→  // Submit
   308→  const handleSubmit = async (e: React.FormEvent) => {
   309→    e.preventDefault();
   310→    if (!validate()) return;
   311→
   312→    setLoading(true);
   313→    try {
   314→      // Split fullName into firstName and lastName for API compatibility
   315→      const nameParts = fullName.trim().split(/\s+/);
   316→      const firstName = nameParts[0] || '';
   317→      const lastName = nameParts.slice(1).join(' ') || firstName;
   318→
   319→      // Upload photo first if selected
   320→      let photoUrl: string | null = null;
   321→      if (photoFile) {
   322→        try {
   323→          const photoFormData = new FormData();
   324→          photoFormData.append('file', photoFile);
   325→          const uploadRes = await fetch('/api/baggage/upload-photo', {
   326→            method: 'POST',
   327→            body: photoFormData,
   328→          });
   329→          if (uploadRes.ok) {
   330→            const uploadData = await uploadRes.json();
   331→            photoUrl = uploadData.photoUrl;
   332→          }
   333→        } catch {
   334→          // Non-blocking, continue without photo
   335→        }
   336→      }
   337→
   338→      const response = await fetch('/api/activate', {
   339→        method: 'POST',
   340→        headers: { 'Content-Type': 'application/json' },
   341→        body: JSON.stringify({
   342→          reference: reference.toUpperCase(),
   343→          travelerFirstName: firstName,
   344→          travelerLastName: lastName,
   345→          whatsappOwner: phone.trim(),
   346→          airlineName: airline,
   347→          flightNumber: flightNumber.trim().toUpperCase(),
   348→          destination,
   349→          hotelName: hotelName.trim() || undefined,
   350→          hotelAddress: hotelAddress.trim() || undefined,
   351→          hotelPhone: hotelPhone.trim() || undefined,
   352→          roomNumber: roomNumber.trim() || undefined,
   353→          departureDate,
   354→          departureTime: departureTime || undefined,
   355→          transportMode: 'flight',
   356→          photoUrl: photoUrl || undefined,
   357→          activationDate: deferredActivation && activationDate ? activationDate : undefined,
   358→        }),
   359→      });
   360→
   361→      if (response.ok) {
   362→        const data = await response.json();
   363→        sessionStorage.setItem('activationData', JSON.stringify({
   364→          reference: reference.toUpperCase(),
   365→          firstName,
   366→          lastName,
   367→          whatsapp: phone.trim(),
   368→          airlineName: airline,
   369→          flightNumber,
   370→          destination,
   371→          hotelName,
   372→          hotelAddress,
   373→          hotelPhone,
   374→          roomNumber,
   375→          type: 'hajj',
   376→          activatedAt: new Date().toISOString(),
   377→          expiresAt: data.baggage?.expiresAt,
   378→          photoUrl: photoUrl || data.baggage?.photoUrl || null,
   379→          // Store base64 preview for instant display on confirmation page (no server request needed)
   380→          photoPreview: photoPreview || null,
   381→        }));
   382→
   383→        const params = new URLSearchParams({
   384→          code: reference.toUpperCase(),
   385→          firstName,
   386→          lastName,
   387→          flight: flightNumber,
   388→          destination,
   389→          chefPhone: phone.trim(),
   390→        });
   391→        if (photoUrl) params.set('photo', photoUrl);
   392→        router.push(`/activate/confirmation?${params.toString()}`);
   393→      } else {
   394→        const error = await response.json();
   395→        toast({ title: error.message || t.errorTitle, variant: 'destructive' });
   396→      }
   397→    } catch {
   398→      toast({ title: t.errorMsg, variant: 'destructive' });
   399→    } finally {
   400→      setLoading(false);
   401→    }
   402→  };
   403→
   404→  // Min date = today
   405→  const today = new Date().toISOString().split('T')[0];
   406→
   407→  return (
   408→    <main dir={dir} className="min-h-screen flex flex-col items-center" style={{ background: BG, color: TEXT, padding: '16px' }}>
   409→
   410→      {/* ─── Header ─── */}
   411→      <div className="w-full max-w-[480px] flex items-center justify-between mb-6 animate-[fadeInDown_0.5s_ease]">
   412→        <div>
   413→          <div className="text-[28px] font-extrabold tracking-tight text-black">
   414→            <span className="text-white bg-black px-2 py-0.5 rounded-md mr-1">Pass</span>Hajj
   415→          </div>
   416→          <div className="text-sm mt-1 text-black/60">{t.subtitle}</div>
   417→        </div>
   418→        <LangSelector lang={lang} setLang={setLang} />
   419→      </div>
   420→
   421→      {/* ─── Progress Bar ─── */}
   422→      <div className="w-full max-w-[480px] mb-5">
   423→        <div className="h-1 bg-white/40 rounded-full overflow-hidden">
   424→          <div className="h-full bg-black rounded-full" style={{ width: '50%' }} />
   425→        </div>
   426→      </div>
   427→
   428→      {/* ─── Form Card ─── */}
   429→      <form onSubmit={handleSubmit} className="w-full max-w-[480px]">
   430→        <div
   431→          className="rounded-[20px] p-7 shadow-[0_8px_24px_rgba(0,0,0,0.12)] mb-4 animate-[fadeInUp_0.5s_ease_0.2s_both]"
   432→          style={{ background: CARD_BG }}
   433→        >
   434→          <div className="text-lg font-bold mb-5 flex items-center gap-2">
   435→            {t.title}
   436→          </div>
   437→
   438→          {/* QR Code (Auto-détecté) */}
   439→          <div className="mb-5">
   440→            <Label className="text-[13px] font-semibold mb-1.5 block">{t.qrCode}</Label>
   441→            <Input
   442→              value={reference}
   443→              onChange={(e) => setReference(e.target.value.toUpperCase())}
   444→              readOnly={!!qrFromUrl}
   445→              placeholder={t.qrPlaceholder}
   446→              className={`h-[52px] rounded-xl text-[15px] ${qrFromUrl ? 'bg-[#f1f5f9] cursor-not-allowed text-gray-600' : 'bg-[#f8fafc]'} border-[#e2e8f0]`}
   447→            />
   448→            {qrFromUrl && (
   449→              <p className="text-xs mt-1.5 text-gray-500">{t.qrDetected}</p>
   450→            )}
   451→          </div>
   452→
   453→          {/* Full Name */}
   454→          <div className="mb-5">
   455→            <Label className="text-[13px] font-semibold mb-1.5 block">{t.fullName}</Label>
   456→            <Input

   458→              onChange={(e) => setFullName(e.target.value)}

   471→              value={phone}
   472→              onChange={setPhone}
   473→              placeholder="77 123 45 67"

   476→            <p className="text-xs mt-1.5 text-gray-500">{t.whatsappHint}</p>

   479→          {/* Email (optional) */}

   513→              placeholder={t.flightPlaceholder}
   514→              required

   517→          </div>
