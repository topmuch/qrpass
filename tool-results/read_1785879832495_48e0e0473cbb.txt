   210→function AISuggestions({ agencyId, stats }: { agencyId: string; stats: Stats }) {

   212→  const [loading, setLoading] = useState(false);
   213→  const [error, setError] = useState<string | null>(null);
   214→  const [dismissedSuggestions, setDismissedSuggestions] = useState<Set<number>>(new Set());
   215→
   216→  const fetchSuggestion = async () => {
   217→    setLoading(true);
   218→    setError(null);
   219→    try {
   220→      const response = await fetch(`/api/ai/suggestions?agencyId=${agencyId}`);
   221→      const data = await response.json();
   222→      if (data.success && data.suggestion) {
   223→        if (typeof data.suggestion === 'object') {
   224→          const s = data.suggestion;
   225→          setSuggestion(`Nous vous recommandons ${s.recommended} QR codes pour votre prochaine campagne. Basé sur: ${s.basedOn}`);
   226→        } else {
   227→          setSuggestion(data.suggestion);
   228→        }
   229→      } else {
   230→        setError('Impossible de charger les suggestions');
   231→      }
   232→    } catch (err) {
   233→      console.error('Error fetching AI suggestion:', err);
   234→      setError('Erreur lors du chargement');
   235→    } finally {
   236→      setLoading(false);
   237→    }
   238→  };
   239→
   240→  useEffect(() => {
   241→    fetchSuggestion();
   242→  }, [agencyId]);
   243→
   244→  const handleDismiss = (index: number) => {
   245→    setDismissedSuggestions(prev => new Set([...prev, index]));
   246→  };
   247→
   248→  const getContextualSuggestions = (): { icon: string; title: string; text: string; color: string }[] => {
   249→    const suggestions: { icon: string; title: string; text: string; color: string }[] = [];

   252→      suggestions.push({
   253→        icon: '⚠️',

   257→      });
   258→    }
   259→
   260→    if (stats.pending > 5) {
   261→      suggestions.push({
   262→        icon: '⏳',
   263→        title: 'Activation en attente',
   264→        text: `${stats.pending} bagages sont en attente d'activation. Envoyez un rappel aux voyageurs.`,
   265→        color: 'bg-amber-50 dark:bg-blue-600/10 border-amber-200 dark:border-amber-800 text-amber-700 dark:text-blue-500'
   266→      });
   267→    }
   268→
   269→    if (stats.scanned > 0) {
   270→      suggestions.push({
   271→        icon: '🔍',
   272→        title: 'Scans récents',
   273→        text: `${stats.scanned} bagage(s) ont été scanné(s) récemment. Vérifiez les localisations.`,
   274→        color: 'bg-blue-50 dark:bg-blue-500/10 border-blue-200 dark:border-blue-800 text-blue-700 dark:text-blue-400'
   275→      });
   276→    }
   277→
   278→    if (stats.total > 0 && stats.pending === 0 && stats.lost === 0) {
   279→      suggestions.push({
   280→        icon: '✅',
   281→        title: 'Excellent !',
   282→        text: 'Tous vos bagages sont actifs et bien suivis. Continuez comme ça !',
   283→        color: 'bg-emerald-50 dark:bg-blue-600/10 border-emerald-200 dark:border-emerald-800 text-emerald-700 dark:text-blue-500'
   284→      });
   285→    }
   286→
   287→    return suggestions;
   288→  };
   289→
   290→  const contextualSuggestions = getContextualSuggestions();
   291→
   292→  return (
   293→    <div className="bg-white dark:bg-slate-900 rounded-2xl p-6 border border-slate-200 dark:border-slate-800">
   294→      <div className="flex items-center justify-between mb-4">
   295→        <div className="flex items-center gap-3">
   296→          <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-blue-500 to-amber-600 flex items-center justify-center">
   297→            <Sparkles className="w-5 h-5 text-white" />
   298→          </div>
   299→          <div>
   300→            <h3 className="text-lg font-semibold text-slate-800 dark:text-white">Suggestions IA</h3>
   301→            <p className="text-sm text-slate-500 dark:text-slate-400">Recommandations personnalisées</p>
   302→          </div>
   303→        </div>
   304→        <button
   305→          onClick={fetchSuggestion}
   306→          disabled={loading}
   307→          className="p-2 rounded-xl hover:bg-slate-100 dark:hover:bg-slate-800 transition-colors"
   308→          title="Actualiser"
   309→        >
   310→          <RefreshCw className={`w-4 h-4 text-slate-500 dark:text-slate-400 ${loading ? 'animate-spin' : ''}`} />
   311→        </button>
   312→      </div>
   313→
   314→      <div className="space-y-3 mb-4">
   315→        {contextualSuggestions.map((sugg, index) => (
   316→          !dismissedSuggestions.has(index) && (
   317→            <div key={index} className={`p-4 rounded-xl border ${sugg.color} relative group`}>
   318→              <div className="flex items-start gap-3">
   319→                <span className="text-xl">{sugg.icon}</span>
   320→                <div className="flex-1">
   321→                  <p className="font-medium">{sugg.title}</p>
   322→                  <p className="text-sm opacity-80">{sugg.text}</p>
   323→                </div>
   324→                <button
   325→                  onClick={() => handleDismiss(index)}
   326→                  className="opacity-0 group-hover:opacity-100 p-1 rounded-lg hover:bg-black/10 dark:hover:bg-white/10 transition-all"
   327→                  title="Masquer cette notification"
   328→                >
   329→                  <X className="w-4 h-4" />
   330→                </button>

   339→          <div className="flex items-center gap-2">
   340→            <Sparkles className="w-4 h-4 text-blue-600 animate-pulse" />
   341→            <span className="text-slate-500 dark:text-slate-400 text-sm">Analyse en cours...</span>
   342→          </div>
   343→        </div>
   344→      )}
   345→
   346→      {suggestion && !loading && (
   347→        <div className="p-4 bg-gradient-to-r from-amber-50 to-blue-50 dark:from-blue-600/10 dark:to-blue-500/10 border border-amber-200 dark:border-amber-800 rounded-xl">

   356→      )}
   357→
   358→      {error && !loading && (
   359→        <div className="p-4 bg-rose-50 dark:bg-rose-500/10 border border-rose-200 dark:border-rose-800 rounded-xl">
   360→          <p className="text-sm text-rose-600 dark:text-rose-400">{error}</p>
   361→        </div>
   362→      )}
   363→    </div>
   364→  );

   367→// Advertisement Banner Component
   368→function AdBanner() {
   369→  const [ads, setAds] = useState<Array<{
   370→    id: string;
   371→    title: string;
   372→    imageUrl: string;
   373→    linkUrl: string | null;
   374→    targetScope: string;
   375→  }>>([]);
   376→  const [currentIndex, setCurrentIndex] = useState(0);
   377→  const [loading, setLoading] = useState(true);
   378→
   379→  useEffect(() => {
   380→    const fetchAds = async () => {
   381→      try {
   382→        const res = await fetch('/api/advertisements');
   383→        const data = await res.json();
   384→        if (data.advertisements && data.advertisements.length > 0) {
   385→          setAds(data.advertisements);
   386→        }
   387→      } catch (err) {
   388→        console.error('Error fetching ads:', err);
   389→      } finally {
   390→        setLoading(false);
   391→      }
   392→    };
   393→    fetchAds();
   394→  }, []);
   395→
   396→  useEffect(() => {
   397→    if (ads.length <= 1) return;
   398→    const timer = setInterval(() => {
   399→      setCurrentIndex(prev => (prev + 1) % ads.length);
   400→    }, 5000);
   401→    return () => clearInterval(timer);
   402→  }, [ads.length]);
   403→
   404→  if (loading || ads.length === 0) return null;
   405→
   406→  const ad = ads[currentIndex];
   407→
   408→  return (
   409→    <div className="relative rounded-2xl overflow-hidden border border-slate-200 dark:border-slate-800 bg-white dark:bg-slate-900">
   410→      {ad.imageUrl ? (
   411→        <a href={ad.linkUrl || '#'} target="_blank" rel="noopener noreferrer" className="block">
   412→          <img src={ad.imageUrl} alt={ad.title} className="w-full h-40 object-cover" />
   413→          {ad.title && (
   414→            <div className="absolute bottom-0 left-0 right-0 bg-gradient-to-t from-black/60 to-transparent p-4">
   415→              <p className="text-white text-sm font-medium">{ad.title}</p>
   416→            </div>
   417→          )}
   418→        </a>
   419→      ) : (
   420→        <a href={ad.linkUrl || '#'} target="_blank" rel="noopener noreferrer" className="block p-6 bg-gradient-to-r from-amber-50 to-blue-50 dark:from-blue-600/10 dark:to-blue-500/10 text-center">
   421→          <p className="text-slate-800 dark:text-white font-semibold">{ad.title}</p>
   422→        </a>
   423→      )}
   424→      {ads.length > 1 && (
   425→        <div className="absolute bottom-2 right-2 flex gap-1">
   426→          {ads.map((_, i) => (
   427→            <button key={i} onClick={() => setCurrentIndex(i)} className={`w-2 h-2 rounded-full transition-all ${i === currentIndex ? 'bg-white w-4' : 'bg-white/50'}`} />
   428→          ))}
   429→        </div>
   430→      )}
   431→    </div>
   432→  );
   433→}
