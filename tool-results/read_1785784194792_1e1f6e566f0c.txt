   212→// ─── Main Component ───
   213→function BaggageActivateContent() {
   214→  const router = useRouter();
   215→  const searchParams = useSearchParams();
   216→  const qrFromUrl = searchParams.get('code') || searchParams.get('qr') || '';
   217→
   218→  // Language
   219→  const [lang, setLang] = useState<Lang>('fr');
   220→  const dir = lang === 'ar' ? 'rtl' : 'ltr';
   221→  const t = T[lang];
   222→
   223→  // Form state
   224→  const [reference, setReference] = useState('');
   225→  const [fullName, setFullName] = useState('');
   226→  const [phone, setPhone] = useState('');
   227→  const [phoneCountry, setPhoneCountry] = useState('SN'); // default Senegal
   228→  const [email, setEmail] = useState('');
   229→  const [airline, setAirline] = useState('');
   230→  const [flightNumber, setFlightNumber] = useState('');
   231→  const [destination, setDestination] = useState('');
   232→  const [departureDate, setDepartureDate] = useState('');
   233→  const [departureTime, setDepartureTime] = useState('');
   234→  const [hotelName, setHotelName] = useState('');
   235→  const [roomNumber, setRoomNumber] = useState('');
   236→  const [photoPreview, setPhotoPreview] = useState<string | null>(null);
   237→  const [photoFile, setPhotoFile] = useState<File | null>(null);
   238→  const [loading, setLoading] = useState(false);
   239→
   240→  // Deferred activation
   241→  const [deferredActivation, setDeferredActivation] = useState(false);
   242→  const [activationDate, setActivationDate] = useState('');
   243→
   244→  // Pre-fill QR from URL
   245→  useEffect(() => {
   246→    if (qrFromUrl) {
   247→      setReference(qrFromUrl.toUpperCase());
   248→    }
   249→  }, [qrFromUrl]);
   250→
   251→  // Detect country from IP for phone input
   252→  useEffect(() => {
   253→    fetch('/api/ip-country')
   254→      .then(res => res.json())
   255→      .then(data => {
   256→        if (data.country) setPhoneCountry(data.country);
   257→      })
   258→      .catch(() => {}); // silent fallback to SN
   259→  }, []);
   260→
   261→  // Photo preview
   262→  const handlePhotoChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   263→    const file = e.target.files?.[0];
   264→    if (file) {
   265→      setPhotoFile(file);
   266→      const reader = new FileReader();
   267→      reader.onload = (ev) => setPhotoPreview(ev.target?.result as string);
   268→      reader.readAsDataURL(file);
   269→    }
   270→  };
   271→
   272→  // Form validation
   273→  const validate = (): boolean => {
   274→    if (!reference.trim()) {
   275→      toast({ title: 'Code QR requis', variant: 'destructive' });
   276→      return false;
   277→    }
   278→    if (!fullName.trim()) {
   279→      toast({ title: 'Nom complet requis', variant: 'destructive' });
   280→      return false;
   281→    }
   282→    if (!phone.trim()) {
   283→      toast({ title: 'Numéro WhatsApp requis', variant: 'destructive' });
   284→      return false;
   285→    }
   286→    if (!airline) {
   287→      toast({ title: 'Compagnie aérienne requise', variant: 'destructive' });
   288→      return false;
   289→    }
   290→    if (!flightNumber.trim()) {
   291→      toast({ title: 'Numéro de vol requis', variant: 'destructive' });
   292→      return false;
   293→    }
   294→    if (!destination) {
   295→      toast({ title: 'Destination requise', variant: 'destructive' });
   296→      return false;
   297→    }
   298→    if (!departureDate) {
   299→      toast({ title: 'Date de départ requise', variant: 'destructive' });
   300→      return false;
   301→    }
   302→    return true;
   303→  };
   304→
   305→  // Submit
   306→  const handleSubmit = async (e: React.FormEvent) => {
   307→    e.preventDefault();
   308→    if (!validate()) return;
   309→
   310→    setLoading(true);
   311→    try {
   312→      // Split fullName into firstName and lastName for API compatibility
   313→      const nameParts = fullName.trim().split(/\s+/);
   314→      const firstName = nameParts[0] || '';
   315→      const lastName = nameParts.slice(1).join(' ') || firstName;
   316→
   317→      // Upload photo first if selected
   318→      let photoUrl: string | null = null;
   319→      if (photoFile) {
   320→        try {
   321→          const photoFormData = new FormData();
   322→          photoFormData.append('file', photoFile);
   323→          const uploadRes = await fetch('/api/baggage/upload-photo', {
   324→            method: 'POST',
   325→            body: photoFormData,
   326→          });
   327→          if (uploadRes.ok) {
   328→            const uploadData = await uploadRes.json();
   329→            photoUrl = uploadData.photoUrl;
   330→          }
   331→        } catch {
   332→          // Non-blocking, continue without photo
   333→        }
   334→      }
   335→
   336→      const response = await fetch('/api/activate', {
   337→        method: 'POST',
   338→        headers: { 'Content-Type': 'application/json' },
   339→        body: JSON.stringify({
   340→          reference: reference.toUpperCase(),
   341→          travelerFirstName: firstName,
   342→          travelerLastName: lastName,
   343→          whatsappOwner: phone.trim(),
   344→          airlineName: airline,
   345→          flightNumber: flightNumber.trim().toUpperCase(),
   346→          destination,
   347→          hotelName: hotelName.trim() || undefined,
   348→          roomNumber: roomNumber.trim() || undefined,
   349→          departureDate,
   350→          departureTime: departureTime || undefined,
   351→          transportMode: 'flight',
   352→          photoUrl: photoUrl || undefined,
   353→          activationDate: deferredActivation && activationDate ? activationDate : undefined,
   354→        }),
   355→      });
   356→
   357→      if (response.ok) {
   358→        const data = await response.json();
   359→        sessionStorage.setItem('activationData', JSON.stringify({
   360→          reference: reference.toUpperCase(),
   361→          firstName,
   362→          lastName,
   363→          whatsapp: phone.trim(),
   364→          airlineName: airline,
   365→          flightNumber,
   366→          destination,
   367→          hotelName,
   368→          roomNumber,
   369→          type: 'hajj',
   370→          activatedAt: new Date().toISOString(),
   371→          expiresAt: data.baggage?.expiresAt,
   372→          photoUrl: photoUrl || data.baggage?.photoUrl || null,
   373→          // Store base64 preview for instant display on confirmation page (no server request needed)
   374→          photoPreview: photoPreview || null,
   375→        }));
   376→
   377→        const params = new URLSearchParams({
   378→          code: reference.toUpperCase(),
   379→          firstName,
   380→          lastName,
   381→          flight: flightNumber,
   382→          destination,
   383→          chefPhone: phone.trim(),
   384→        });
   385→        if (photoUrl) params.set('photo', photoUrl);
   386→        router.push(`/activate/confirmation?${params.toString()}`);
   387→      } else {
   388→        const error = await response.json();
   389→        toast({ title: error.message || t.errorTitle, variant: 'destructive' });
   390→      }
   391→    } catch {
   392→      toast({ title: t.errorMsg, variant: 'destructive' });
   393→    } finally {
   394→      setLoading(false);
   395→    }
   396→  };
   397→
   398→  // Min date = today
   399→  const today = new Date().toISOString().split('T')[0];
   400→
   401→  return (
   402→    <main dir={dir} className="min-h-screen flex flex-col items-center" style={{ background: BG, color: TEXT, padding: '16px' }}>
   403→
   404→      {/* ─── Header ─── */}
   405→      <div className="w-full max-w-[480px] flex items-center justify-between mb-6 animate-[fadeInDown_0.5s_ease]">
   406→        <div>
   407→          <div className="text-[28px] font-extrabold tracking-tight text-black">
   408→            <span className="text-white bg-black px-2 py-0.5 rounded-md mr-1">Pass</span>Hajj
   409→          </div>
   410→          <div className="text-sm mt-1 text-black/60">{t.subtitle}</div>
   411→        </div>
   412→        <LangSelector lang={lang} setLang={setLang} />
   413→      </div>
   414→
   415→      {/* ─── Progress Bar ─── */}
   416→      <div className="w-full max-w-[480px] mb-5">
   417→        <div className="h-1 bg-white/40 rounded-full overflow-hidden">
   418→          <div className="h-full bg-black rounded-full" style={{ width: '50%' }} />
   419→        </div>
   420→      </div>
   421→
   422→      {/* ─── Form Card ─── */}
   423→      <form onSubmit={handleSubmit} className="w-full max-w-[480px]">
   424→        <div
   425→          className="rounded-[20px] p-7 shadow-[0_8px_24px_rgba(0,0,0,0.12)] mb-4 animate-[fadeInUp_0.5s_ease_0.2s_both]"
   426→          style={{ background: CARD_BG }}
   427→        >
   428→          <div className="text-lg font-bold mb-5 flex items-center gap-2">
   429→            {t.title}
   430→          </div>
   431→
   432→          {/* QR Code (Auto-détecté) */}
   433→          <div className="mb-5">
   434→            <Label className="text-[13px] font-semibold mb-1.5 block">{t.qrCode}</Label>
   435→            <Input
   436→              value={reference}
   437→              onChange={(e) => setReference(e.target.value.toUpperCase())}
   438→              readOnly={!!qrFromUrl}
   439→              placeholder={t.qrPlaceholder}
   440→              className={`h-[52px] rounded-xl text-[15px] ${qrFromUrl ? 'bg-[#f1f5f9] cursor-not-allowed text-gray-600' : 'bg-[#f8fafc]'} border-[#e2e8f0]`}
   441→            />
   442→            {qrFromUrl && (
   443→              <p className="text-xs mt-1.5 text-gray-500">{t.qrDetected}</p>
   444→            )}
   445→          </div>
   446→
   447→          {/* Full Name */}
   448→          <div className="mb-5">
   449→            <Label className="text-[13px] font-semibold mb-1.5 block">{t.fullName}</Label>
   450→            <Input
   451→              value={fullName}
   452→              onChange={(e) => setFullName(e.target.value)}
   453→              placeholder={t.fullNamePlaceholder}
   454→              required
   455→              className={inputCls}
   456→            />
   457→          </div>
   458→
   459→          {/* WhatsApp - PhoneInput with country flag & auto IP detection */}
   460→          <div className="mb-5">
   461→            <Label className="text-[13px] font-semibold mb-1.5 block">{t.whatsapp}</Label>
   462→            <PhoneInput
   463→              countryCode={phoneCountry}
   464→              onCountryChange={setPhoneCountry}
   465→              value={phone}
   466→              onChange={setPhone}
   467→              placeholder="77 123 45 67"
   468→              required
   469→            />
   470→            <p className="text-xs mt-1.5 text-gray-500">{t.whatsappHint}</p>
   471→          </div>
   472→
   473→          {/* Email (optional) */}
   474→          <div className="mb-5">
   475→            <Label className="text-[13px] font-semibold mb-1.5 block">{t.email}</Label>
   476→            <Input
   477→              type="email"
   478→              value={email}
   479→              onChange={(e) => setEmail(e.target.value)}
   480→              placeholder={t.emailPlaceholder}
   481→              className={inputCls}
   482→            />
   483→          </div>
   484→
   485→          {/* Airline - SELECT DROPDOWN */}
   486→          <div className="mb-5">
   487→            <Label className="text-[13px] font-semibold mb-1.5 block">{t.airline}</Label>
   488→            <select
   489→              value={airline}
   490→              onChange={(e) => setAirline(e.target.value)}
   491→              required
   492→              className={`w-full ${inputCls} appearance-none cursor-pointer`}
   493→            >

   495→              {AIRLINES.map((a) => (

   498→            </select>

   501→          {/* Flight Number */}

   504→            <Input
   505→              value={flightNumber}
   506→              onChange={(e) => setFlightNumber(e.target.value.toUpperCase())}
   507→              placeholder={t.flightPlaceholder}
   508→              required
   509→              className={`${inputCls} font-mono tracking-wider`}
   510→            />
   511→          </div>
   512→
   513→          {/* Destination */}
   514→          <div className="mb-5">
   515→            <Label className="text-[13px] font-semibold mb-1.5 block">{t.destination}</Label>
   516→            <select
   517→              value={destination}
   518→              onChange={(e) => setDestination(e.target.value)}
   519→              required
